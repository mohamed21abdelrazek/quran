     1→"use client"
     2→
     3→import { useEffect, useMemo, useState } from "react"
     4→import { Button } from "@/components/ui/button"
     5→import { Card, CardContent } from "@/components/ui/card"
     6→import { Badge } from "@/components/ui/badge"
     7→import { ProgressRing } from "./progress-ring"
     8→import { fireConfetti } from "@/hooks/use-app"
     9→import type { AppState, TaskDTO } from "@/lib/types"
    10→import type { Quality } from "@/lib/content"
    11→import { dateAr, formatDays, formatPages, weekdayShortAr } from "@/lib/format"
    12→import { cn } from "@/lib/utils"
    13→import { useToast } from "@/hooks/use-toast"
    14→import {
    15→  AlertDialog,
    16→  AlertDialogAction,
    17→  AlertDialogCancel,
    18→  AlertDialogContent,
    19→  AlertDialogDescription,
    20→  AlertDialogFooter,
    21→  AlertDialogHeader,
    22→  AlertDialogTitle,
    23→  AlertDialogTrigger,
    24→} from "@/components/ui/alert-dialog"
    25→import {
    26→  Flame,
    27→  Trophy,
    28→  CheckCheck,
    29→  Check,
    30→  AlertTriangle,
    31→  Clock,
    32→  BookOpen,
    33→  Sparkles,
    34→  PartyPopper,
    35→  CalendarDays,
    36→  Hourglass,
    37→  Star,
    38→  ShieldCheck,
    39→  Crosshair,
    40→  Wand2,
    41→  Loader2,
    42→  Eye,
    43→} from "lucide-react"
    44→
    45→interface ViewProps {
    46→  state: AppState
    47→  load: () => Promise<void>
    48→  action: (url: string, body: unknown) => Promise<Record<string, unknown>>
    49→  openWird: (from: number, to: number, label: string) => void
    50→  patchTask: (taskId: number, status: "done" | "snoozed", quality?: TaskDTO["quality"]) => void
    51→}
    52→
    53→export function DashboardView({ state, load, action, openWird, patchTask }: ViewProps) {
    54→  const { toast } = useToast()
    55→  const [busyTask, setBusyTask] = useState<number | null>(null)
    56→  const [applyingCapacity, setApplyingCapacity] = useState(false)
    57→
    58→  const plan = state.plan
    59→  const tasks = plan?.tasks ?? []
    60→  const pendingTasks = tasks.filter((t) => t.status === "pending")
    61→  const finishedTasks = tasks.filter((t) => t.status !== "pending")
    62→  const donePages = tasks.filter((t) => t.status === "done").reduce((s, t) => s + t.pages, 0)
    63→  const planPages = tasks.reduce((s, t) => s + t.pages, 0)
    64→  const allDone = tasks.length > 0 && tasks.every((t) => t.status === "done")
    65→
    66→  const hour = new Date().getHours()
    67→  const greeting =
    68→    hour < 5 ? "أسعد الله ليلتك" : hour < 12 ? "صباح الخير" : hour < 17 ? "طاب يومك" : "مساء الخير"
    69→
    70→  const hijriDate = useMemo(() => {
    71→    try {
    72→      return new Intl.DateTimeFormat("ar-EG-u-ca-islamic-umalqura", {
    73→        day: "numeric",
    74→        month: "long",
    75→        year: "numeric",
    76→      }).format(new Date())
    77→    } catch {
    78→      return ""
    79→    }
    80→  }, [])
    81→
    82→  const complete = async (task: TaskDTO, quality: Quality) => {
    83→    setBusyTask(task.id)
    84→    try {
    85→      const res = await action("/api/task", { taskId: task.id, action: "complete", quality })
    86→      if (res.queued) {
    87→        // أوفلاين: نثبّتها محلياً وترسل تلقائياً عند عودة الاتصال
    88→        patchTask(task.id, "done", quality)
    89→        toast({
    90→          title: "حُفظت أوفلاين",
    91→          description: "سُجلت مراجعتك وستُرسل تلقائياً فور عودة الاتصال",
    92→        })
    93→        return
    94→      }
    95→      if (res.allDone) {
    96→        fireConfetti()
    97→        toast({
    98→          title: "تقبل الله منك!",
    99→          description: "أنجزت ورد اليوم كاملاً — أحسنت وبارك الله فيك",
   100→        })
   101→      } else {
   102→        const msgs: Record<Quality, string> = {
   103→          perfect: "ما شاء الله! رفعنا قوة هذه الصفحات",
   104→          good: "أحسنت! سنراها مرة أخرى في وقتها",
   105→          hard: "لا بأس — سنراجعها قريباً لتثبت أكثر",
   106→        }
   107→        toast({ title: "سُجلت المراجعة", description: msgs[quality] })
   108→      }
   109→      await load()
   110→    } catch (e) {
   111→      toast({ title: "خطأ", description: e instanceof Error ? e.message : "حاول مرة أخرى" })
   112→    } finally {
   113→      setBusyTask(null)
   114→    }
   115→  }
   116→
   117→  const snooze = async (task: TaskDTO) => {
   118→    setBusyTask(task.id)
   119→    try {
   120→      const res = await action("/api/task", { taskId: task.id, action: "snooze" })
   121→      if (res.queued) {
   122→        patchTask(task.id, "snoozed")
   123→        toast({ title: "حُفظت أوفلاين", description: "التأجيل سيُرسل تلقائياً فور عودة الاتصال" })
   124→        return
   125→      }
   126→      toast({ title: "أجّلناها إلى الغد", description: "ستكون أول ما يظهر في ورد الغد إن شاء الله" })
   127→      await load()
   128→    } catch (e) {
   129→      toast({ title: "خطأ", description: e instanceof Error ? e.message : "حاول مرة أخرى" })
   130→    } finally {
   131→      setBusyTask(null)
   132→    }
   133→  }
   134→
   135→  const cannotToday = async () => {
   136→    try {
   137→      const res = await action("/api/day", { action: "cannot" })
   138→      if (res.queued) {
   139→        toast({ title: "حُفظت أوفلاين", description: "سيُنفذ النقل تلقائياً فور عودة الاتصال" })
   140→        return
   141→      }
   142→      toast({
   143→        title: "لا مشكلة أبداً",
   144→        description: "أعدنا جدولة ورد اليوم ليوم غد، وسنوازن خطتك تلقائياً بما يناسبك",
   145→      })
   146→      await load()
   147→    } catch (e) {
   148→      toast({ title: "خطأ", description: e instanceof Error ? e.message : "حاول مرة أخرى" })
   149→    }
   150→  }
   151→
   152→  const applySuggestedCapacity = async () => {
   153→    setApplyingCapacity(true)
   154→    try {
   155→      const res = await action("/api/settings", { dailyCapacity: state.cycle.requiredDaily })
   156→      if (res.queued) {
   157→        toast({ title: "حُفظت أوفلاين", description: "سيُطبق الضبط تلقائياً فور عودة الاتصال" })
   158→        return
   159→      }
   160→      toast({
   161→        title: "تم ضبط سعتك",
   162→        description: `أصبح وردك ${state.cycle.requiredDaily} صفحة يومياً — تكمل كل حفظك كل ${formatDays(state.cycle.targetDays)} بإذن الله`,
   163→      })
   164→      await load()
   165→    } catch (e) {
   166→      toast({ title: "خطأ", description: e instanceof Error ? e.message : "حاول مرة أخرى" })
   167→    } finally {
   168→      setApplyingCapacity(false)
   169→    }
   170→  }
   171→
   172→  return (
   173→    <div className="space-y-5">
   174→      {/* بطاقة الترحيب */}
   175→      <Card className="overflow-hidden border-0 bg-gradient-to-bl from-emerald-800 via-emerald-700 to-teal-800 text-white glow-emerald">
   176→        <div className="pattern-islamic-light">
   177→          <CardContent className="flex flex-col gap-5 p-5 sm:flex-row sm:items-center sm:justify-between sm:p-7">
   178→            <div className="space-y-3 text-center sm:text-right">
   179→              <div>
   180→                <h2 className="text-2xl font-extrabold sm:text-3xl">
   181→                  {greeting} {state.profile.name || "يا حافظ القرآن"}
   182→                </h2>
   183→                <p className="mt-1.5 flex flex-wrap items-center justify-center gap-x-2 gap-y-1 text-xs text-white/75 sm:justify-start sm:text-sm">
   184→                  <span className="inline-flex items-center gap-1">
   185→                    <CalendarDays className="h-3.5 w-3.5" />
   186→                    {dateAr(state.today)}
   187→                  </span>
   188→                  {hijriDate && <span className="text-gold">• {hijriDate}</span>}
   189→                </p>
   190→              </div>
   191→              <div className="flex flex-wrap items-center justify-center gap-2 sm:justify-start">
   192→                <span className="inline-flex items-center gap-1.5 rounded-full bg-white/15 px-3 py-1.5 text-xs font-bold backdrop-blur-sm">
   193→                  <Flame className="h-4 w-4 text-amber-300" />
   194→                  {state.profile.streak > 0 ? `${state.profile.streak} يوم متتابع` : "ابدأ سلسلتك اليوم"}
   195→                </span>
   196→                <span className="inline-flex items-center gap-1.5 rounded-full bg-white/15 px-3 py-1.5 text-xs font-bold backdrop-blur-sm">
   197→                  <Star className="h-4 w-4 text-amber-300" />
   198→                  {state.level.name} • {state.profile.xp} نقطة
   199→                </span>
   200→              </div>
   201→            </div>
   202→            <div className="flex shrink-0 justify-center">
   203→              <ProgressRing
   204→                value={donePages}
   205→                max={planPages}
   206→                size={132}
   207→                label={planPages > 0 ? `${donePages}/${planPages}` : "—"}
   208→                sublabel="من ورد اليوم"
   209→              />
   210→            </div>
   211→          </CardContent>
   212→        </div>
   213→      </Card>
   214→
   215→      {/* بطاقة دورة الأمان — قاعدة عدم النسيان */}
   216→      {state.cycle.memorizedTotal > 0 && (
   217→        <Card
   218→          className={cn(
   219→            "animate-in fade-in-50 border shadow-sm",
   220→            state.cycle.onTrack
   221→              ? "border-emerald-300/70 bg-emerald-50/70 dark:border-emerald-700/50 dark:bg-emerald-950/30"
   222→              : "border-amber-300/70 bg-amber-50/70 dark:border-amber-600/40 dark:bg-amber-950/30"
   223→          )}
   224→        >
   225→          <CardContent className="space-y-3.5 p-4 sm:p-5">
   226→            <div className="flex flex-wrap items-center justify-between gap-2">
   227→              <div className="flex items-center gap-2">
   228→                <span
   229→                  className={cn(
   230→                    "flex h-9 w-9 items-center justify-center rounded-xl",
   231→                    state.cycle.onTrack
   232→                      ? "bg-emerald-600/15 text-emerald-700 dark:text-emerald-300"
   233→                      : "bg-amber-500/15 text-amber-700 dark:text-amber-300"
   234→                  )}
   235→                >
   236→                  <ShieldCheck className="h-5 w-5" />
   237→                </span>
   238→                <div>
   239→                  <p className="text-sm font-extrabold">دورة الأمان — كل حفظك خلال {formatDays(state.cycle.targetDays)}</p>
   240→                  <p className="text-[11px] text-muted-foreground">
   241→                    قاعدة عدم النسيان: لا صفحة تنتظر أكثر من {state.cycle.targetDays} أيام
   242→                  </p>
   243→                </div>
   244→              </div>
   245→              <span
   246→                className={cn(
   247→                  "rounded-full px-3 py-1 text-xs font-extrabold tabular-nums",
   248→                  state.cycle.onTrack
   249→                    ? "bg-emerald-600 text-white"
   250→                    : "bg-amber-500 text-white"
   251→                )}
   252→              >
   253→                {Math.round(state.cycle.coveragePct * 100)}%
   254→              </span>
   255→            </div>
   256→
   257→            {/* شريط التغطية */}
   258→            <div>
   259→              <div
   260→                className="h-2.5 overflow-hidden rounded-full bg-white/60 dark:bg-black/25"
   261→                role="progressbar"
   262→                aria-valuenow={Math.round(state.cycle.coveragePct * 100)}
   263→                aria-valuemin={0}
   264→                aria-valuemax={100}
   265→                aria-label="نسبة الحفظ المراجَع خلال دورة الأمان"
   266→              >
   267→                <div
   268→                  className={cn(
   269→                    "h-full rounded-full transition-all duration-700",
   270→                    state.cycle.onTrack
   271→                      ? "bg-gradient-to-l from-emerald-600 to-teal-500"
   272→                      : "bg-gradient-to-l from-amber-500 to-orange-400"
   273→                  )}
   274→                  style={{ width: `${Math.max(3, Math.round(state.cycle.coveragePct * 100))}%` }}
   275→                />
   276→              </div>
   277→              <p className="mt-1.5 text-xs leading-5 text-muted-foreground">
   278→                راجعت <span className="font-bold text-foreground">{state.cycle.coveredPages}</span> من {" "}
   279→                <span className="font-bold text-foreground">{state.cycle.memorizedTotal}</span> صفحة محفوظة خلال آخر{" "}
   280→                {formatDays(state.cycle.targetDays)}
   281→              </p>
   282→            </div>
   283→
   284→            {/* الحالة والتوجيه */}
   285→            {state.cycle.onTrack ? (
   286→              <p className="flex items-start gap-2 text-xs leading-6 text-emerald-800 dark:text-emerald-200/90">
   287→                <Check className="mt-0.5 h-4 w-4 shrink-0" />
   288→                أنت على المسار الصحيح — دورتك الكاملة {formatDays(state.cycle.currentCycleDays)} ووردك دائماً في حدود قدرتك.
   289→              </p>
   290→            ) : (
   291→              <div className="space-y-2.5 rounded-xl border border-amber-300/60 bg-white/70 p-3 dark:border-amber-600/30 dark:bg-black/20">
   292→                <p className="text-xs leading-6 text-amber-900 dark:text-amber-100/90">
   293→                  بسعتك الحالية (<span className="font-bold">{state.profile.dailyCapacity}</span> صفحة/يوم) دورتك الكاملة ≈{" "}
   294→                  <span className="font-bold">{formatDays(state.cycle.currentCycleDays)}</span> — لضمان عدم النسيان راجع{" "}
   295→                  <span className="font-bold">{state.cycle.requiredDaily}</span> صفحة يومياً.
   296→                </p>
   297→                <p className="text-[11px] leading-5 text-muted-foreground">
   298→                  لا تقلق: وردك يبدأ دائماً بالأعلى خطراً على النسيان، فالضعيف لا يفلت أبداً مهما كانت سعتك.
   299→                </p>
   300→                <Button
   301→                  size="sm"
   302→                  onClick={applySuggestedCapacity}
   303→                  disabled={applyingCapacity}
   304→                  className="h-9 gap-1.5 bg-amber-600 text-white hover:bg-amber-700"
   305→                >
   306→                  {applyingCapacity ? <Loader2 className="h-4 w-4 animate-spin" /> : <Wand2 className="h-4 w-4" />}
   307→                  اضبط سعتي تلقائياً ({state.cycle.requiredDaily} صفحة/يوم)
   308→                </Button>
   309→              </div>
   310→            )}
   311→          </CardContent>
   312→        </Card>
   313→      )}
   314→
   315→      {/* احتفال اكتمال اليوم */}
   316→      {allDone && (
   317→        <Card className="animate-in fade-in-50 zoom-in-95 border-emerald-300/60 bg-emerald-50 dark:border-emerald-700/50 dark:bg-emerald-950/40">
   318→          <CardContent className="flex items-center gap-4 p-5">
   319→            <div className="flex h-14 w-14 shrink-0 items-center justify-center rounded-2xl bg-gradient-to-br from-amber-400 to-amber-600 text-white">
   320→              <PartyPopper className="h-7 w-7" />
   321→            </div>
   322→            <div>
   323→              <p className="text-base font-extrabold text-emerald-900 dark:text-emerald-200">
   324→                أنجزت ورد اليوم كاملاً — تقبل الله منك!
   325→              </p>
   326→              <p className="mt-1 text-sm text-emerald-700 dark:text-emerald-300/80">
   327→                كسبت 30 نقطة إضافية. استراحة مستحقة، ونراك غداً في موعدك مع كتاب الله.
   328→              </p>
   329→            </div>
   330→          </CardContent>
   331→        </Card>
   332→      )}
   333→
   334→      {/* تنبيه المتأخرات */}
   335→      {state.overdueCount > 0 && !allDone && (
   336→        <Card className="border-amber-300/70 bg-amber-50/80 dark:border-amber-600/40 dark:bg-amber-950/30">
   337→          <CardContent className="flex items-start gap-3 p-4">
   338→            <Hourglass className="mt-0.5 h-5 w-5 shrink-0 text-amber-600 dark:text-amber-400" />
   339→            <div className="text-sm leading-6">
   340→              <span className="font-bold text-amber-900 dark:text-amber-200">
   341→                لديك {formatPages(state.overdueCount)} متأخرة عن موعدها.
   342→              </span>{" "}
   343→              <span className="text-amber-800/80 dark:text-amber-300/80">
   344→                وردك يبدأ دائماً بالأعلى خطورة على النسيان — فالأكثر تأخراً والأضعف يظهر لك أولاً حتى لا يفلت شيء.
   345→              </span>
   346→            </div>
   347→          </CardContent>
   348→        </Card>
   349→      )}
   350→
   351→      {/* ورد اليوم */}
   352→      <section aria-label="ورد اليوم" className="space-y-3">
   353→        <div className="flex items-center justify-between">
   354→          <h3 className="flex items-center gap-2 text-lg font-extrabold">
   355→            <BookOpen className="h-5 w-5 text-primary" />
   356→            ورد اليوم
   357→          </h3>
   358→          {plan && (
   359→            <Badge variant="secondary" className="gap-1 font-bold">
   360→              <Sparkles className="h-3.5 w-3.5 text-gold" />
   361→              سعة اليوم: {plan.capacity} صفحات
   362→            </Badge>
   363→          )}
   364→        </div>
   365→
   366→        {tasks.length === 0 && (
   367→          <Card>
   368→            <CardContent className="flex flex-col items-center gap-3 p-8 text-center">
   369→              <BookOpen className="h-10 w-10 text-muted-foreground/40" />
   370→              <p className="text-sm text-muted-foreground">
   371→                لا توجد مهام اليوم. أضف صفحات محفوظة من الإعدادات ليبدأ وِردي بتخطيط مراجعاتك.
   372→              </p>
   373→            </CardContent>
   374→          </Card>
   375→        )}
   376→
   377→        {/* المهام المعلقة */}
   378→        <div className="space-y-3">
   379→          {pendingTasks.map((task) => (
   380→            <TaskCard
   381→              key={task.id}
   382→              task={task}
   383→              busy={busyTask === task.id}
   384→              onComplete={complete}
   385→              onSnooze={snooze}
   386→              onOpen={(t) => openWird(t.pageFrom, t.pageTo, t.label)}
   387→            />
   388→          ))}
   389→        </div>
   390→
   391→        {/* المهام المنتهية */}
   392→        {finishedTasks.length > 0 && (
   393→          <div className="space-y-2 pt-2">
   394→            <p className="text-xs font-bold text-muted-foreground">ما تم اليوم</p>
   395→            <div className="space-y-2">
   396→              {finishedTasks.map((task) => (
   397→                <FinishedTaskRow key={task.id} task={task} />
   398→              ))}
   399→            </div>
   400→          </div>
   401→        )}
   402→
   403→        {/* زر تعذر المراجعة */}
   404→        {pendingTasks.length > 0 && (
   405→          <AlertDialog>
   406→            <AlertDialogTrigger asChild>
   407→              <Button variant="ghost" className="w-full text-muted-foreground" size="sm">
   408→                <AlertTriangle className="h-4 w-4" />
   409→                لم أستطع المراجعة اليوم
   410→              </Button>
   411→            </AlertDialogTrigger>
   412→            <AlertDialogContent>
   413→              <AlertDialogHeader>
   414→                <AlertDialogTitle>لم تستطع المراجعة اليوم؟</AlertDialogTitle>
   415→                <AlertDialogDescription>
   416→                  لا مشكلة أبداً — الديمومة أهم من الكمال. سنغلق ورد اليوم وننقل صفحاته إلى الغد، وسيتكيف النظام مع وتيرتك
   417→                  تلقائياً.
   418→                </AlertDialogDescription>
   419→              </AlertDialogHeader>
   420→              <AlertDialogFooter>
   421→                <AlertDialogCancel>سأراجع شيئاً على الأقل</AlertDialogCancel>
   422→                <AlertDialogAction onClick={cannotToday}>نعم، انقلها للغد</AlertDialogAction>
   423→              </AlertDialogFooter>
   424→            </AlertDialogContent>
   425→          </AlertDialog>
   426→        )}
   427→      </section>
   428→
   429→      {/* الصفحات المحتاجة مراجعة — تركيز على الأضعف */}
   430→      {state.cycle.memorizedTotal > 0 && (
   431→        <section aria-label="الصفحات المحتاجة مراجعة" className="space-y-3">
   432→          <div className="flex items-center justify-between">
   433→            <h3 className="flex items-center gap-2 text-lg font-extrabold">
   434→              <Crosshair className="h-5 w-5 text-primary" />
   435→              تحتاج انتباهك
   436→            </h3>
   437→            {state.attention.length > 0 && (
   438→              <Badge variant="secondary" className="text-[11px]">
   439→                <Eye className="h-3 w-3" />
   440→                الأعلى خطورة على النسيان
   441→              </Badge>
   442→            )}
   443→          </div>
   444→
   445→          {state.attention.length === 0 ? (
