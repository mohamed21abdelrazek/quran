    50→
    51→export function DashboardView({ state, load, action }: ViewProps) {
    52→  const { toast } = useToast()
    53→  const [busyTask, setBusyTask] = useState<number | null>(null)
    54→  const [applyingCapacity, setApplyingCapacity] = useState(false)
    55→
    56→  const plan = state.plan
    57→  const tasks = plan?.tasks ?? []
    58→  const pendingTasks = tasks.filter((t) => t.status === "pending")
    59→  const finishedTasks = tasks.filter((t) => t.status !== "pending")
    60→  const donePages = tasks.filter((t) => t.status === "done").reduce((s, t) => s + t.pages, 0)
    61→  const planPages = tasks.reduce((s, t) => s + t.pages, 0)
    62→  const allDone = tasks.length > 0 && tasks.every((t) => t.status === "done")
    63→
    64→  const hour = new Date().getHours()
    65→  const greeting =
    66→    hour < 5 ? "أسعد الله ليلتك" : hour < 12 ? "صباح الخير" : hour < 17 ? "طاب يومك" : "مساء الخير"
    67→
    68→  const hijriDate = useMemo(() => {
    69→    try {
    70→      return new Intl.DateTimeFormat("ar-EG-u-ca-islamic-umalqura", {
    71→        day: "numeric",
    72→        month: "long",
    73→        year: "numeric",
    74→      }).format(new Date())
    75→    } catch {
    76→      return ""
    77→    }
    78→  }, [])
    79→
    80→  const complete = async (task: TaskDTO, quality: Quality) => {
    81→    setBusyTask(task.id)

    83→      const res = await action("/api/task", { taskId: task.id, action: "complete", quality })
    84→      if (res.allDone) {
    85→        fireConfetti()
    86→        toast({
    87→          title: "تقبل الله منك!",
    88→          description: "أنجزت ورد اليوم كاملاً — أحسنت وبارك الله فيك",
    89→        })
    90→      } else {
    91→        const msgs: Record<Quality, string> = {
    92→          perfect: "ما شاء الله! رفعنا قوة هذه الصفحات",
    93→          good: "أحسنت! سنراها مرة أخرى في وقتها",
    94→          hard: "لا بأس — سنراجعها قريباً لتثبت أكثر",
    95→        }
    96→        toast({ title: "سُجلت المراجعة", description: msgs[quality] })
    97→      }
    98→      await load()
    99→    } catch (e) {
   100→      toast({ title: "خطأ", description: e instanceof Error ? e.message : "حاول مرة أخرى" })
   101→    } finally {
   102→      setBusyTask(null)
   103→    }
   104→  }
   105→
   106→  const snooze = async (task: TaskDTO) => {
   107→    setBusyTask(task.id)
   108→    try {
   109→      await action("/api/task", { taskId: task.id, action: "snooze" })
   110→      toast({ title: "أجّلناها إلى الغد", description: "ستكون أول ما يظهر في ورد الغد إن شاء الله" })
   111→      await load()
   112→    } catch (e) {
   113→      toast({ title: "خطأ", description: e instanceof Error ? e.message : "حاول مرة أخرى" })
   114→    } finally {
   115→      setBusyTask(null)
   116→    }
   117→  }
   118→
   119→  const cannotToday = async () => {
   120→    try {
   121→      await action("/api/day", { action: "cannot" })
   122→      toast({
   123→        title: "لا مشكلة أبداً",
   124→        description: "أعدنا جدولة ورد اليوم ليوم غد، وسنوازن خطتك تلقائياً بما يناسبك",
   125→      })
   126→      await load()
   127→    } catch (e) {
   128→      toast({ title: "خطأ", description: e instanceof Error ? e.message : "حاول مرة أخرى" })
   129→    }
   130→  }
   131→
   132→  const applySuggestedCapacity = async () => {
   133→    setApplyingCapacity(true)
   134→    try {
   135→      await action("/api/settings", { dailyCapacity: state.cycle.requiredDaily })
   136→      toast({
   137→        title: "تم ضبط سعتك",
   138→        description: `أصبح وردك ${state.cycle.requiredDaily} صفحة يومياً — تكمل كل حفظك كل ${formatDays(state.cycle.targetDays)} بإذن الله`,
   139→      })
   140→      await load()
   141→    } catch (e) {
   142→      toast({ title: "خطأ", description: e instanceof Error ? e.message : "حاول مرة أخرى" })
   143→    } finally {
   144→      setApplyingCapacity(false)
   145→    }
   146→  }
   147→
   148→  return (
   149→    <div className="space-y-5">
   150→      {/* بطاقة الترحيب */}
   151→      <Card className="overflow-hidden border-0 bg-gradient-to-bl from-emerald-800 via-emerald-700 to-teal-800 text-white glow-emerald">
   152→        <div className="pattern-islamic-light">
   153→          <CardContent className="flex flex-col gap-5 p-5 sm:flex-row sm:items-center sm:justify-between sm:p-7">
   154→            <div className="space-y-3 text-center sm:text-right">
   155→              <div>
   156→                <h2 className="text-2xl font-extrabold sm:text-3xl">
   157→                  {greeting} {state.profile.name || "يا حافظ القرآن"}
   158→                </h2>
   159→                <p className="mt-1.5 flex flex-wrap items-center justify-center gap-x-2 gap-y-1 text-xs text-white/75 sm:justify-start sm:text-sm">
   160→                  <span className="inline-flex items-center gap-1">
   161→                    <CalendarDays className="h-3.5 w-3.5" />
   162→                    {dateAr(state.today)}
   163→                  </span>
   164→                  {hijriDate && <span className="text-gold">• {hijriDate}</span>}
   165→                </p>
   166→              </div>
   167→              <div className="flex flex-wrap items-center justify-center gap-2 sm:justify-start">
   168→                <span className="inline-flex items-center gap-1.5 rounded-full bg-white/15 px-3 py-1.5 text-xs font-bold backdrop-blur-sm">
   169→                  <Flame className="h-4 w-4 text-amber-300" />
   170→                  {state.profile.streak > 0 ? `${state.profile.streak} يوم متتابع` : "ابدأ سلسلتك اليوم"}
   171→                </span>
   172→                <span className="inline-flex items-center gap-1.5 rounded-full bg-white/15 px-3 py-1.5 text-xs font-bold backdrop-blur-sm">
   173→                  <Star className="h-4 w-4 text-amber-300" />
   174→                  {state.level.name} • {state.profile.xp} نقطة
   175→                </span>
   176→              </div>
   177→            </div>
   178→            <div className="flex shrink-0 justify-center">
   179→              <ProgressRing
   180→                value={donePages}
   181→                max={planPages}
   182→                size={132}
   183→                label={planPages > 0 ? `${donePages}/${planPages}` : "—"}
   184→                sublabel="من ورد اليوم"
   185→              />
   186→            </div>

   188→        </div>
   189→      </Card>
   190→

   193→        <Card
   194→          className={cn(
   195→            "animate-in fade-in-50 border shadow-sm",
   196→            state.cycle.onTrack
   197→              ? "border-emerald-300/70 bg-emerald-50/70 dark:border-emerald-700/50 dark:bg-emerald-950/30"
   198→              : "border-amber-300/70 bg-amber-50/70 dark:border-amber-600/40 dark:bg-amber-950/30"
   199→          )}
   200→        >
   201→          <CardContent className="space-y-3.5 p-4 sm:p-5">
   202→            <div className="flex flex-wrap items-center justify-between gap-2">
   203→              <div className="flex items-center gap-2">
   204→                <span
   205→                  className={cn(
   206→                    "flex h-9 w-9 items-center justify-center rounded-xl",
   207→                    state.cycle.onTrack
   208→                      ? "bg-emerald-600/15 text-emerald-700 dark:text-emerald-300"
   209→                      : "bg-amber-500/15 text-amber-700 dark:text-amber-300"
   210→                  )}
   211→                >
   212→                  <ShieldCheck className="h-5 w-5" />
   213→                </span>
   214→                <div>
   215→                  <p className="text-sm font-extrabold">دورة الأمان — كل حفظك خلال {formatDays(state.cycle.targetDays)}</p>
   216→                  <p className="text-[11px] text-muted-foreground">
   217→                    قاعدة عدم النسيان: لا صفحة تنتظر أكثر من {state.cycle.targetDays} أيام
   218→                  </p>
   219→                </div>
   220→              </div>
   221→              <span
   222→                className={cn(
   223→                  "rounded-full px-3 py-1 text-xs font-extrabold tabular-nums",
   224→                  state.cycle.onTrack
   225→                    ? "bg-emerald-600 text-white"
   226→                    : "bg-amber-500 text-white"
   227→                )}
   228→              >
   229→                {Math.round(state.cycle.coveragePct * 100)}%
   230→              </span>
   231→            </div>
   232→
   233→            {/* شريط التغطية */}
   234→            <div>
   235→              <div
   236→                className="h-2.5 overflow-hidden rounded-full bg-white/60 dark:bg-black/25"
   237→                role="progressbar"
   238→                aria-valuenow={Math.round(state.cycle.coveragePct * 100)}
   239→                aria-valuemin={0}
   240→                aria-valuemax={100}
   241→                aria-label="نسبة الحفظ المراجَع خلال دورة الأمان"
   242→              >
   243→                <div
   244→                  className={cn(
   245→                    "h-full rounded-full transition-all duration-700",
   246→                    state.cycle.onTrack
   247→                      ? "bg-gradient-to-l from-emerald-600 to-teal-500"
   248→                      : "bg-gradient-to-l from-amber-500 to-orange-400"
   249→                  )}
   250→                  style={{ width: `${Math.max(3, Math.round(state.cycle.coveragePct * 100))}%` }}
   251→                />
   252→              </div>
   253→              <p className="mt-1.5 text-xs leading-5 text-muted-foreground">
   254→                راجعت <span className="font-bold text-foreground">{state.cycle.coveredPages}</span> من {" "}
   255→                <span className="font-bold text-foreground">{state.cycle.memorizedTotal}</span> صفحة محفوظة خلال آخر{" "}
   256→                {formatDays(state.cycle.targetDays)}
   257→              </p>
   258→            </div>
   259→
   260→            {/* الحالة والتوجيه */}
   261→            {state.cycle.onTrack ? (
   262→              <p className="flex items-start gap-2 text-xs leading-6 text-emerald-800 dark:text-emerald-200/90">
   263→                <Check className="mt-0.5 h-4 w-4 shrink-0" />
   264→                أنت على المسار الصحيح — دورتك الكاملة {formatDays(state.cycle.currentCycleDays)} ووردك دائماً في حدود قدرتك.
   265→              </p>
   266→            ) : (
   267→              <div className="space-y-2.5 rounded-xl border border-amber-300/60 bg-white/70 p-3 dark:border-amber-600/30 dark:bg-black/20">
   268→                <p className="text-xs leading-6 text-amber-900 dark:text-amber-100/90">
   269→                  بسعتك الحالية (<span className="font-bold">{state.profile.dailyCapacity}</span> صفحة/يوم) دورتك الكاملة ≈{" "}
   270→                  <span className="font-bold">{formatDays(state.cycle.currentCycleDays)}</span> — لضمان عدم النسيان راجع{" "}
   271→                  <span className="font-bold">{state.cycle.requiredDaily}</span> صفحة يومياً.
   272→                </p>
   273→                <p className="text-[11px] leading-5 text-muted-foreground">
   274→                  لا تقلق: وردك يبدأ دائماً بالأعلى خطراً على النسيان، فالضعيف لا يفلت أبداً مهما كانت سعتك.
   275→                </p>
   276→                <Button
   277→                  size="sm"
   278→                  onClick={applySuggestedCapacity}
   279→                  disabled={applyingCapacity}
   280→                  className="h-9 gap-1.5 bg-amber-600 text-white hover:bg-amber-700"
   281→                >
   282→                  {applyingCapacity ? <Loader2 className="h-4 w-4 animate-spin" /> : <Wand2 className="h-4 w-4" />}
   283→                  اضبط سعتي تلقائياً ({state.cycle.requiredDaily} صفحة/يوم)
   284→                </Button>
   285→              </div>
   286→            )}
   287→          </CardContent>
   288→        </Card>
   289→      )}
   290→
   291→      {/* احتفال اكتمال اليوم */}
   292→      {allDone && (
   293→        <Card className="animate-in fade-in-50 zoom-in-95 border-emerald-300/60 bg-emerald-50 dark:border-emerald-700/50 dark:bg-emerald-950/40">
   294→          <CardContent className="flex items-center gap-4 p-5">
   295→            <div className="flex h-14 w-14 shrink-0 items-center justify-center rounded-2xl bg-gradient-to-br from-amber-400 to-amber-600 text-white">
   296→              <PartyPopper className="h-7 w-7" />
   297→            </div>
   298→            <div>
   299→              <p className="text-base font-extrabold text-emerald-900 dark:text-emerald-200">
   300→                أنجزت ورد اليوم كاملاً — تقبل الله منك!
   301→              </p>
   302→              <p className="mt-1 text-sm text-emerald-700 dark:text-emerald-300/80">
   303→                كسبت 30 نقطة إضافية. استراحة مستحقة، ونراك غداً في موعدك مع كتاب الله.
   304→              </p>
   305→            </div>
   306→          </CardContent>
   307→        </Card>
   308→      )}
   309→
   310→      {/* تنبيه المتأخرات */}
   311→      {state.overdueCount > 0 && !allDone && (
   312→        <Card className="border-amber-300/70 bg-amber-50/80 dark:border-amber-600/40 dark:bg-amber-950/30">
   313→          <CardContent className="flex items-start gap-3 p-4">
   314→            <Hourglass className="mt-0.5 h-5 w-5 shrink-0 text-amber-600 dark:text-amber-400" />
   315→            <div className="text-sm leading-6">
   316→              <span className="font-bold text-amber-900 dark:text-amber-200">
   317→                لديك {formatPages(state.overdueCount)} متأخرة عن موعدها.
   318→              </span>{" "}
   319→              <span className="text-amber-800/80 dark:text-amber-300/80">
   320→                وردك يبدأ دائماً بالأعلى خطورة على النسيان — فالأكثر تأخراً والأضعف يظهر لك أولاً حتى لا يفلت شيء.
   321→              </span>
   322→            </div>
   323→          </CardContent>
   324→        </Card>
   325→      )}
   326→
   327→      {/* ورد اليوم */}
   328→      <section aria-label="ورد اليوم" className="space-y-3">
   329→        <div className="flex items-center justify-between">
   330→          <h3 className="flex items-center gap-2 text-lg font-extrabold">
   331→            <BookOpen className="h-5 w-5 text-primary" />
   332→            ورد اليوم
   333→          </h3>
   334→          {plan && (
   335→            <Badge variant="secondary" className="gap-1 font-bold">
   336→              <Sparkles className="h-3.5 w-3.5 text-gold" />
   337→              سعة اليوم: {plan.capacity} صفحات
   338→            </Badge>

   340→        </div>
   341→
   342→        {tasks.length === 0 && (
   343→          <Card>
   344→            <CardContent className="flex flex-col items-center gap-3 p-8 text-center">
   345→              <BookOpen className="h-10 w-10 text-muted-foreground/40" />
   346→              <p className="text-sm text-muted-foreground">
   347→                لا توجد مهام اليوم. أضف صفحات محفوظة من الإعدادات ليبدأ وِردي بتخطيط مراجعاتك.
   348→              </p>
   349→            </CardContent>
   350→          </Card>
   351→        )}
   352→
   353→        {/* المهام المعلقة */}
   354→        <div className="space-y-3">
   355→          {pendingTasks.map((task) => (
   356→            <TaskCard
   357→              key={task.id}
   358→              task={task}
   359→              busy={busyTask === task.id}
   360→              onComplete={complete}
   361→              onSnooze={snooze}
   362→            />
   363→          ))}
   364→        </div>
   365→
   366→        {/* المهام المنتهية */}
   367→        {finishedTasks.length > 0 && (
   368→          <div className="space-y-2 pt-2">
   369→            <p className="text-xs font-bold text-muted-foreground">ما تم اليوم</p>

   371→              {finishedTasks.map((task) => (
   372→                <FinishedTaskRow key={task.id} task={task} />
   373→              ))}
   374→            </div>
   375→          </div>
   376→        )}
   377→
   378→        {/* زر تعذر المراجعة */}
   379→        {pendingTasks.length > 0 && (
   380→          <AlertDialog>
   381→            <AlertDialogTrigger asChild>
   382→              <Button variant="ghost" className="w-full text-muted-foreground" size="sm">
   383→                <AlertTriangle className="h-4 w-4" />
   384→                لم أستطع المراجعة اليوم
   385→              </Button>
   386→            </AlertDialogTrigger>
   387→            <AlertDialogContent>
   388→              <AlertDialogHeader>
   389→                <AlertDialogTitle>لم تستطع المراجعة اليوم؟</AlertDialogTitle>
   390→                <AlertDialogDescription>
   391→                  لا مشكلة أبداً — الديمومة أهم من الكمال. سنغلق ورد اليوم وننقل صفحاته إلى الغد، وسيتكيف النظام مع وتيرتك
   392→                  تلقائياً.
   393→                </AlertDialogDescription>
   394→              </AlertDialogHeader>
   395→              <AlertDialogFooter>
   396→                <AlertDialogCancel>سأراجع شيئاً على الأقل</AlertDialogCancel>
   397→                <AlertDialogAction onClick={cannotToday}>نعم، انقلها للغد</AlertDialogAction>
   398→              </AlertDialogFooter>
   399→            </AlertDialogContent>
   400→          </AlertDialog>

   402→      </section>
   403→
   404→      {/* الصفحات المحتاجة مراجعة — تركيز على الأضعف */}

   411→            </h3>
   412→            {state.attention.length > 0 && (
   413→              <Badge variant="secondary" className="text-[11px]">

   416→              </Badge>
   417→            )}
   418→          </div>
   419→
   420→          {state.attention.length === 0 ? (
   421→            <Card className="border-emerald-300/60 bg-emerald-50/70 dark:border-emerald-700/40 dark:bg-emerald-950/25">
   422→              <CardContent className="flex items-center gap-3 p-4">
   423→                <span className="flex h-10 w-10 shrink-0 items-center justify-center rounded-xl bg-emerald-600/15 text-emerald-700 dark:text-emerald-300">
   424→                  <ShieldCheck className="h-5 w-5" />

   436→                  أولاً في الأيام القادمة.

   439→                  {state.attention.map((a) => {
