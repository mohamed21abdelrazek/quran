     6→/// Real-time recitation inspector & e-learning studio sidebar for Windows desktop (Left Bar).
     7→/// Displays:
     8→/// 1. Tajweed errors per word (with phonetic breakdown)
     9→/// 2. Missed & misread words per word
    10→/// 3. Live performance statistics & surah progress
    11→/// 4. Upcoming AI feature telemetry cards (Makhaarij, Hifz Retention, Confidence)
    12→class FeedbackSidebar extends StatefulWidget {
    13→  final HighlightingController controller;
    14→  final bool isRecording;
    15→  final bool isLoadingEngine;
    16→  final VoidCallback? onToggleRecord;
    17→  final VoidCallback? onClearBuffer;
    18→  final VoidCallback? onTajweedToggle;
    19→  final double? width;
    20→  final VoidCallback? onClose;
    21→
    22→  const FeedbackSidebar({
    23→    super.key,
    24→    required this.controller,
    25→    required this.isRecording,
    26→    this.isLoadingEngine = false,
    27→    this.onToggleRecord,
    28→    this.onClearBuffer,
    29→    this.onTajweedToggle,
    30→    this.width,
    31→    this.onClose,
    32→  });
    33→
    34→  @override
    35→  State<FeedbackSidebar> createState() => _FeedbackSidebarState();
    36→}
    37→
    38→class _FeedbackSidebarState extends State<FeedbackSidebar> {
    39→  @override
    40→  void initState() {
    41→    super.initState();
    42→    widget.controller.addListener(_onUpdate);
    43→    widget.controller.activeAyah.addListener(_onUpdate);
    44→    widget.controller.globalRevision.addListener(_onUpdate);
    45→  }
    46→
    47→  @override
    48→  void didUpdateWidget(FeedbackSidebar oldWidget) {
    49→    super.didUpdateWidget(oldWidget);
    50→    if (oldWidget.controller != widget.controller) {
    51→      oldWidget.controller.removeListener(_onUpdate);
    52→      oldWidget.controller.activeAyah.removeListener(_onUpdate);
    53→      oldWidget.controller.globalRevision.removeListener(_onUpdate);
    54→      widget.controller.addListener(_onUpdate);
    55→      widget.controller.activeAyah.addListener(_onUpdate);
    56→      widget.controller.globalRevision.addListener(_onUpdate);
    57→    }
    58→  }
    59→
    60→  @override
    61→  void dispose() {
    62→    widget.controller.removeListener(_onUpdate);
    63→    widget.controller.activeAyah.removeListener(_onUpdate);
    64→    widget.controller.globalRevision.removeListener(_onUpdate);
    65→    super.dispose();
    66→  }
    67→
    68→  void _onUpdate() {
    69→    if (mounted) setState(() {});

    72→  @override
    73→  Widget build(BuildContext context) {
    74→    final app = AppState.instance;
    75→    final c = app.colors;
    76→    final isAr = app.isArabic;
    77→    final ctrl = widget.controller;
    78→
    79→    final targetSurah = ctrl.targetSurah;
    80→    final surahs = ctrl.repository.surahMetadata;
    81→    final surahMeta = (targetSurah >= 1 && targetSurah <= surahs.length) ? surahs[targetSurah - 1] : null;
    82→    final surahName = surahMeta?.surahName ?? 'سورة $targetSurah';
    83→    final totalAyahs = AppConstants.getSurahAyahCount(targetSurah);
    84→
    85→    final activeAyahNum = ctrl.activeAyah.value ?? 1;
    86→    final displayVerses = ctrl.repository.getSurah(targetSurah);
    87→    final activeVerse = displayVerses.cast<QuranVerse?>().firstWhere(
    88→          (v) => v?.ayah == activeAyahNum,
    89→          orElse: () => null,
    90→        );
    91→
    92→    // 1. Collect Tajweed and Phonemic Errors for active verse
    93→    final activeTajweedErrors = <_WordErrorInfo>[];
    94→    // 2. Collect Missed / Misread words (marked Red) for active verse
    95→    final missedWords = <_MissedWordInfo>[];
    96→
    97→    int correctWords = 0;
    98→    int redWords = 0;
    99→    int yellowWords = 0;
   100→
   101→    if (activeVerse != null) {
   102→      for (int i = 0; i < activeVerse.uthmaniWords.length; i++) {
   103→        final w = activeVerse.uthmaniWords[i];
   104→        final isGreen = ctrl.isWordGreen(activeAyahNum, i);
   105→        final isRed = ctrl.isWordRed(activeAyahNum, i);
   106→        final isYellow = ctrl.isWordYellow(activeAyahNum, i);
   107→
   108→        if (isGreen) correctWords++;
   109→        if (isRed) {
   110→          redWords++;
   111→          missedWords.add(_MissedWordInfo(
   112→            word: w,
   113→            wordIndex: i,
   114→            ayah: activeAyahNum,
   115→          ));
   116→        }
   117→        if (isYellow) yellowWords++;
   118→
   119→        final errs = ctrl.getWordErrors(activeAyahNum, i);
   120→        if (errs != null && errs.isNotEmpty) {
   121→          activeTajweedErrors.add(_WordErrorInfo(
   122→            word: w,
   123→            wordIndex: i,
   124→            ayah: activeAyahNum,
   125→            errors: errs,
   126→          ));
   127→        }
   128→      }
   129→    }
   130→

   558→  Widget _buildTajweedErrorCard(String word, ReciterError e, ThemeColors c, bool isAr) {
   559→    final exp = e.expectedPh.isEmpty ? '—' : ArabicUtils.cleanPhonetic(e.expectedPh);
   560→    final pred = e.predictedPh.isEmpty ? '—' : ArabicUtils.cleanPhonetic(e.predictedPh);
   561→
   562→    final (String title, Color accent, String desc) = switch (e.errorType) {
   563→      ErrorCategory.tajweed => (
   564→        (isAr ? e.expectedRule?.name.ar : e.expectedRule?.name.en) ?? (isAr ? 'حكم تجويد' : 'Tajweed'),
   565→        const Color(0xFFF5B738),
   566→        isAr ? 'لم يطبق الحكم بالشكل الصحيح' : 'Rule not applied properly',
   567→      ),
   568→      ErrorCategory.tashkeel => (
   569→        isAr ? 'خطأ في التشكيل' : 'Vowel Discrepancy',
   570→        const Color(0xFF60A5FA),
   571→        isAr ? 'الحركة تختلف عن الصحيحة' : 'Vowel differs',
   572→      ),
   573→      ErrorCategory.normal => (
   574→        isAr ? 'خطأ في النطق' : 'Pronunciation',
   575→        c.red,
   576→        isAr ? 'نطق الحرف يختلف عن النص' : 'Letter differs',
   577→      ),
   578→    };
   579→
   580→    return Container(
   581→      padding: const EdgeInsets.all(10),
   582→      decoration: BoxDecoration(
   583→        color: c.surfaceHigh,
   584→        borderRadius: BorderRadius.circular(14),
   585→        border: Border.all(color: accent.withValues(alpha: 0.35)),
   586→      ),
   587→      child: Column(
   588→        crossAxisAlignment: CrossAxisAlignment.start,
   589→        children: [
   590→          Row(
   591→            mainAxisAlignment: MainAxisAlignment.spaceBetween,
   592→            children: [
   593→              Text(
   594→                word,
   595→                style: TextStyle(
   596→                  fontFamily: AppConstants.hafsFontFamily,
   597→                  color: c.text,
   598→                  fontSize: 16,
   599→                  fontWeight: FontWeight.bold,
   600→                ),
   601→              ),
   602→              Container(
   603→                padding: const EdgeInsets.symmetric(horizontal: 6, vertical: 2),
   604→                decoration: BoxDecoration(
   605→                  color: accent.withValues(alpha: 0.15),
   606→                  borderRadius: BorderRadius.circular(6),
   607→                ),
   608→                child: Text(
   609→                  title,
