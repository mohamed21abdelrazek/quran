   105→double _availableHeight(BuildContext context) {
   106→  // top,bottom padding, will include notch and stuff
   107→  final mqPadding = MediaQuery.paddingOf(context);
   108→  final viewPadding = View.of(context).viewPadding;
   109→  final dpr = MediaQuery.devicePixelRatioOf(context);
   110→  final double top = (viewPadding.top / dpr) + mqPadding.top;
   111→  final double bottom = (viewPadding.bottom / dpr) + mqPadding.bottom;
   112→
   113→  final dualPage = Settings.instance.temporaryState.dualPage;
   114→
   115→  final appBarHeight = dualPage ? 0 : kToolbarHeight;
   116→  final padding = top + bottom + appBarHeight;
   117→
   118→  // dont go below 200, we will scroll if below
   119→  return max(200, MediaQuery.sizeOf(context).height - (padding));
   120→}

   482→  bool _isMutashabihaAyat(int surahAyahIdx, int surahIdx) {
   483→    // _mutashabihat is sorted
   484→    final it = _mutashabihat
   485→        .skipWhile((m) => m.src.surahIdx < surahIdx)
   486→        .takeWhile(
   487→          (m) =>
   488→              m.src.surahIdx == surahIdx &&
   489→              m.src.surahAyahIndexes.first <= surahAyahIdx,
   490→        );
   491→    for (final m in it) {
   492→      if (m.src.surahAyahIndexes.contains(surahAyahIdx)) {
   493→        return true;
   494→      }
   495→    }
   496→    return false;
   497→  }

   514→  static Future<Translation> _loadTranslation() async {
   515→    ByteBuffer transUtf8;
   516→    final isBundledTranslation = Settings.instance.translationFile.isEmpty;
   517→    if (isBundledTranslation) {
   518→      transUtf8 = (await rootBundle.load("assets/ur.junagarhi.txt")).buffer;
   519→    } else {
   520→      final data = File(Settings.instance.translationFile).readAsBytesSync();
   521→      transUtf8 = data.buffer;
   522→    }
   523→
   524→    List<int> transLineOffsets = [];
   525→    transLineOffsets.add(0);
   526→    int start = 0;
   527→    final utf = transUtf8.asUint8List();
   528→    int next = utf.indexOf(10);
   529→    while (next != -1) {
   530→      transLineOffsets.add(next);
   531→
   532→      start = next + 1;
   533→      next = utf.indexOf(10, start);
   534→    }
   535→
   536→    return Translation(
   537→      fileName: Settings.instance.translationFile,
   538→      transUtf8: transUtf8,
   539→      transLineOffsets: transLineOffsets,
   540→      isBundledTranslation: isBundledTranslation,
   541→    );
   542→  }

   544→  void _onAyahLongPressed(int ayahIdx) async {
   545→    int surahIdx = surahForAyah(ayahIdx);
   546→    int surahAyah = toSurahAyahOffset(surahIdx, ayahIdx);
   547→    List<Mutashabiha> mutashabihat = _getMutashabihaAyat(surahAyah, surahIdx);
   548→
   549→    _translation ??= await _loadTranslation();
   550→
   551→    for (int i = 0; i < mutashabihat.length; ++i) {
   552→      mutashabihat[i].loadText();
   553→    }
   554→
   555→    Widget? mutashabihaWidget;
   556→    if (mutashabihat.isNotEmpty) {
   557→      mutashabihaWidget = ListView.separated(
   558→        physics: const NeverScrollableScrollPhysics(),
   559→        shrinkWrap: true,
   560→        separatorBuilder: (ctx, index) => const Divider(height: 1),
   561→        itemCount: mutashabihat.length,
   562→        itemBuilder: (ctx, index) {
   563→          return MutashabihaAyatListItem(
   564→            mutashabiha: mutashabihat[index],
   565→            onGotoMutashabiha: (int ayah) {
   566→              final page = getPageForAyah(ayah, Settings.instance.mushaf);
   567→              Navigator.of(context).pushNamed(
   568→                goToPageModal,
   569→                arguments: ReadOnlyQuranPageArgs(widget.model, page),
   570→              );
   571→            },
   572→          );
   573→        },
   574→      );
   575→    }
   576→
   577→    if (!mounted) return;
   578→
   579→    return await showModalBottomSheet(
   580→      context: context,
   581→      scrollControlDisabledMaxHeightRatio:
   582→          Settings.instance.temporaryState.dualPage ? 0.8 : 0.6,
   583→      builder: (context) {
   584→        return SizedBox(
   585→          width: MediaQuery.sizeOf(context).width,
   586→          child: Padding(
   587→            padding: const EdgeInsets.only(top: 8.0, bottom: 8.0),
   588→            child: LongPressActionSheet(
   589→              mutashabihaList: mutashabihaWidget,
   590→              translation: _translation!,
   591→              tappedAyahIdx: ayahIdx,
   592→            ),
   593→          ),
   594→        );
   595→      },
   596→    );
   597→  }

   599→  void _onAyahTapped(int ayahIdx, int wordIdx, bool longPress) async {
   600→    bool tapToShowBottomSheet = Settings.instance.tapToShowTranslation;
   601→    if ((longPress && !tapToShowBottomSheet) ||
   602→        (!longPress && tapToShowBottomSheet)) {
   603→      _onAyahLongPressed(ayahIdx);
   604→      return;
   605→    }
   606→
   607→    Ayat? ayatInDb = _getAyatInDB(ayahIdx);
   608→    // otherwise we add/remove ayah
   609→    if (ayatInDb != null && ayatInDb.markedWords.contains(wordIdx)) {
   610→      // remove
   611→      widget.model.removeMarkedWordInAyat(ayahIdx, wordIdx);
   612→    } else {
   613→      // add
   614→      Ayat ayat = Ayat("", [wordIdx], ayahIdx: ayahIdx);
   615→      widget.model.addAyahs([ayat]);
   616→    }
   617→  }

   619→  static List<Line> _linesForPage(List<layout.Page> pages, int pageIndex) {
   620→    final page = pages[pageIndex];
   621→    final List<Line> pageLines = List.filled(
   622→      page.lines.length,
   623→      Line(const []),
   624→      growable: false,
   625→    );
   626→    int currentLine = 0;
   627→
   628→    for (
   629→      int lineIdx = 0;
   630→      lineIdx < page.lines.length;
   631→      ++lineIdx, ++currentLine
   632→    ) {
   633→      final l = page.lines[lineIdx];
   634→      final ayah = l.ayahIdx;
   635→      final start = l.wordStartInAyahIdx;
   636→
   637→      if (ayah < 0) {
   638→        pageLines[currentLine] = Line([
   639→          LineAyah(start == -999 ? start : start - 1, ""),
   640→        ]);
   641→        continue;
   642→      }
   643→
   644→      int? nextAyah;
   645→      int? nextAyahStart;
   646→      if (lineIdx + 1 < page.lines.length) {
   647→        if (page.lines[lineIdx + 1].ayahIdx >= 0) {
   648→          final nextLine = page.lines[lineIdx + 1];
   649→          nextAyah = nextLine.ayahIdx;
   650→          nextAyahStart = nextLine.wordStartInAyahIdx;
   651→        } else {
   652→          // find next valid ayah
   653→          for (int j = lineIdx + 1; j < page.lines.length; ++j) {
   654→            if (page.lines[j].ayahIdx >= 0) {
   655→              nextAyah = page.lines[j].ayahIdx;
   656→              nextAyahStart = 0;
   657→              break;
   658→            } else if (page.lines[j].ayahIdx < 0) {
   659→              // we are at the end of surah. Check if there are any ayahs left
   660→              int surah = surahForAyah(ayah);
   661→              assert(surah >= 0);
   662→              int surahAyah = toSurahAyahOffset(surah, ayah);
   663→              final surahData = surahDataForIdx(surah);
   664→              if (surahAyah + 1 < surahData.ayahCount) {
   665→                // go to the end of surah
   666→                nextAyah = toAbsoluteAyahOffset(surah, surahData.ayahCount);
   667→                nextAyahStart = 0;
   668→              }
   669→            }
   670→          }
   671→        }
   672→      } else if (pageIndex + 1 < pages.length) {
   673→        final nextPage = pages[pageIndex + 1];
   674→        if (nextPage.lines.first.ayahIdx >= 0) {
   675→          final nextPageFirstLine = nextPage.lines.first;
   676→          nextAyah = nextPageFirstLine.ayahIdx;
   677→          nextAyahStart = nextPageFirstLine.wordStartInAyahIdx;
   678→        } else {
   679→          // find next valid ayah
   680→          for (int j = 0; j < nextPage.lines.length; ++j) {
   681→            if (nextPage.lines[j].ayahIdx >= 0) {
   682→              nextAyah = nextPage.lines[j].ayahIdx;
   683→              nextAyahStart = 0;
   684→              break;
   685→            }
   686→          }
   687→        }
   688→      }
   689→      // last page, second last ayah
   690→      else if (pageIndex + 1 == pages.length &&
   691→          ayah == 6234 &&
   692→          lineIdx + 1 >= page.lines.length) {
   693→        // Do a +1 so that it adds the whole last ayah
   694→        // ayahsForRanges adds all ayahs (ayah, nextAyah]
   695→        // and for the nextAyah it adds upto nextAyahStart
   696→        nextAyah = 6235 + 1;
   697→        nextAyahStart = 0;
   698→      }
   699→
   700→      final ayahsForRanges = QuranText.instance.ayahsForRanges(
   701→        ayah,
   702→        start,
   703→        nextAyah,
   704→        nextAyahStart,
   705→      );
   706→      List<LineAyah> lineAyahs = ayahsForRanges
   707→          .map((a) => LineAyah(a.$1, a.$2))
   708→          .toList(growable: false);
   709→      pageLines[currentLine] = Line(lineAyahs);
   710→    }
   711→    return pageLines;
   712→  }

   978→  // Finds the correct position of the first word of the line
   979→  // in the full ayah text
   980→  static int getFirstWordIndex(
   981→    String fullAyah,
   982→    String currentLine,
   983→    int startSearchAt,
   984→  ) {
   985→    // Find the position of currentLine in full Ayah
   986→    int match = fullAyah.indexOf(currentLine, startSearchAt);
   987→    if (match == -1) {
   988→      throw "Didn't find anything, bug!\n$currentLine\n$fullAyah";
   989→    }
   990→
   991→    // Find the index of first word
   992→    int s = fullAyah.indexOf("\u200c");
   993→    int wordIndex = 0;
   994→    while (true) {
   995→      if (s >= match) break;
   996→      s = fullAyah.indexOf("\u200c", s + 1);
   997→      wordIndex++;
   998→    }
   999→    return wordIndex;
  1000→  }

  1007→  List<TextSpan> _buildLineSpans(
  1008→    Line line,
  1009→    int lineIdx,
  1010→    List<(int ayahIndex, int, int, Ayat?, bool, String)> ayahData,
  1011→    ThemeData themeData, {
  1012→    required bool reflowMode,
  1013→  }) {
  1014→    List<TextSpan> spans = [];
  1015→    final isIndoPk = isIndoPak(Settings.instance.mushaf);
  1016→    bool darkMode = themeData.brightness == Brightness.dark;
  1017→
  1018→    for (final a in line.lineAyahs) {
  1019→      final (
  1020→        _,
  1021→        int surahIdx,
  1022→        int surahAyahIdx,
  1023→        Ayat? ayahInDb,
  1024→        bool isMutashabihaAyat,
  1025→        String fullAyahText,
  1026→      ) = ayahData.firstWhere(
  1027→        (data) {
  1028→          return data.$1 == a.ayahIndex;
  1029→        },
  1030→        orElse: () {
  1031→          throw "Unexpected unable to find the index for given ayahIdx: ${a.ayahIndex}";
  1032→        },
  1033→      );
  1034→
  1035→      String text = a.text;
  1036→
  1037→      int startSearchAt = 0;
  1038→      if (_lastAyahIdx == a.ayahIndex) {
  1039→        // start search in the full ayah text after the
  1040→        // last rendered word
  1041→        startSearchAt = _lastRenderedWordIdx;
  1042→      }
  1043→      _lastAyahIdx = a.ayahIndex;
  1044→
  1045→      List<String> words = text.split('\u200c'); // zwj
  1046→      int i = getFirstWordIndex(fullAyahText, text, startSearchAt);
  1047→
  1048→      int lastWordInLineIndex = words.length - 1;
  1049→      if (words.last.isEmpty) {
  1050→        lastWordInLineIndex -= 1;
  1051→      }
  1052→
  1053→      for (final (idx, w) in words.indexed) {
  1054→        if (w.isEmpty) {
  1055→          i++;
  1056→          continue;
  1057→        }
  1058→
  1059→        int wordIdx = i;
  1060→        TextStyle? style;
  1061→
  1062→        if (w.contains("\u06dd")) {
  1063→          if (isIndoPk) {
  1064→            final kind = isPortion(a.ayahIndex);
  1065→            if (kind != PortionKind.none) {
  1066→              glyphForPortionKind(kind);
  1067→              spans.add(
  1068→                TextSpan(
  1069→                  text: glyphForPortionKind(kind),
  1070→                  style: TextStyle(
  1071→                    backgroundColor:
  1072→                        (darkMode
  1073→                            ? Colors.lightGreen.withAlpha(180)
  1074→                            : Colors.lightGreen),
  1075→                  ),
  1076→                ),
  1077→              );
  1078→            }
  1079→          }
  1080→
  1081→          final ruku = isIndoPk ? getRukuData(a.ayahIndex) : null;
  1082→          final hasRukuMarker = ruku != null;
  1083→          spans.add(
  1084→            TextSpan(
  1085→              text: hasRukuMarker ? " $w" : w,
  1086→              recognizer:
  1087→                  hasRukuMarker
  1088→                      ? _addRecognizer(
  1089→                        TapGestureRecognizer()
  1090→                          ..onTap = () => _onRukuTapped(a.ayahIndex),
  1091→                      )
  1092→                      : null,
  1093→              style: TextStyle(
  1094→                inherit: true,
  1095→                color: themeData.textTheme.bodyMedium?.color,
  1096→                backgroundColor:
  1097→                    hasRukuMarker
  1098→                        ? (darkMode
  1099→                            ? Colors.amber.shade700.withAlpha(125)
  1100→                            : Colors.amber.shade100)
  1101→                        : null,
  1102→              ),
  1103→            ),
  1104→          );
  1105→          i++;
  1106→          continue;
  1107→        }
  1108→
  1109→        if (ayahInDb != null) {
  1110→          if (ayahInDb.markedWords.contains(wordIdx)) {
  1111→            style = _markedWordStyle(darkMode);
  1112→          } else if (isMutashabihaAyat) {
  1113→            style = _markedMutAyahBGStyle(darkMode);
  1114→          } else {
  1115→            style = _markedAyahBGStyle(darkMode);
  1116→          }
  1117→        } else if (isMutashabihaAyat) {
  1118→          style = _mutStyle(darkMode);
  1119→        }
  1120→
  1121→        if (widget.ayahsToHighlight?.contains(a.ayahIndex) ?? false) {
  1122→          if (style != null) {
  1123→            style = style.merge(_highlightStyle(darkMode));
  1124→          } else {
  1125→            style = _highlightStyle(darkMode);
  1126→          }
  1127→        }
  1128→
  1129→        final tapHandler = _addRecognizer(
  1130→          TapAndLongPressGestureRecognizer(
  1131→            onTap: () => widget.onAyahTapped(a.ayahIndex, wordIdx, false),
  1132→            onLongPress: () => widget.onAyahTapped(a.ayahIndex, wordIdx, true),
  1133→          ),
  1134→        );
  1135→
  1136→        // word
  1137→        spans.add(TextSpan(recognizer: tapHandler, text: w, style: style));
  1138→        // space
  1139→        if (reflowMode || idx != lastWordInLineIndex) {
  1140→          spans.add(TextSpan(text: ' ', style: style));
  1141→        }
  1142→
  1143→        i++;
  1144→      }
  1145→
  1146→      _lastRenderedWordIdx = i;
  1147→    }
  1148→    return spans;
  1149→  }

  1161→  double _getWordSpacing(
  1162→    int lineIdx,
  1163→    List<TextSpan> spans,
  1164→    double width,
  1165→    TextStyle style,
  1166→    Mushaf mushaf,
  1167→  ) {
  1168→    final textPainter = TextPainter(
  1169→      text: TextSpan(children: spans, style: style),
  1170→      textDirection: TextDirection.rtl,
  1171→      maxLines: 1,
  1172→    );
  1173→    textPainter.layout();
  1174→    var diffW = (min(width, 700) - textPainter.width);
  1175→    textPainter.dispose();
  1176→
  1177→    if (diffW > 10) {
  1178→      if (isSurahLastLine(widget.pageIndex, lineIdx, mushaf)) {
  1179→        return 2;
  1180→      }
  1181→      int spaces = 0;
  1182→      for (final s in spans) {
  1183→        if (s.text == ' ') spaces++;
  1184→      }
  1185→      return (spaces > 0 ? diffW / (spaces) : 1).roundToDouble();
  1186→    } else {
  1187→      diffW = diffW.abs();
  1188→      if (diffW < 10) {
  1189→        return 2;
  1190→      } else if (diffW < 20) {
  1191→        return 1;
  1192→      } else {
  1193→        return 0;
  1194→      }
  1195→    }
  1196→  }

  1375→  List<Widget> _pageLines(double rowHeight, double rowWidth) {
  1376→    int lastAyah = -1;
  1377→    List<(int, int, int, Ayat?, bool, String)> ayahData = [];
  1378→    ayahData.length = 0;
  1379→    final colorMutashabihat = Settings.instance.colorMutashabihat;
  1380→
  1381→    for (final l in widget._pageLines) {
  1382→      if (l.lineAyahs.first.ayahIndex < 0) {
  1383→        continue; // bismillah
  1384→      }
  1385→      for (final a in l.lineAyahs) {
  1386→        if (lastAyah == a.ayahIndex) {
  1387→          continue;
  1388→        } else {
  1389→          lastAyah = a.ayahIndex;
  1390→          final int surahIdx = surahForAyah(a.ayahIndex);
  1391→          final int surahAyahIdx = toSurahAyahOffset(surahIdx, a.ayahIndex);
  1392→          final Ayat? ayahInDb = widget.getAyatInDB(a.ayahIndex);
  1393→          final text = QuranText.instance.ayahText(a.ayahIndex);
  1394→          final bool isMutashabihaAyat =
  1395→              colorMutashabihat
  1396→                  ? widget.isMutashabihaAyat(surahAyahIdx, surahIdx)
  1397→                  : false;
  1398→
  1399→          ayahData.add((
  1400→            a.ayahIndex,
  1401→            surahIdx,
  1402→            surahAyahIdx,
  1403→            ayahInDb,
  1404→            isMutashabihaAyat,
  1405→            text,
  1406→          ));
  1407→        }
  1408→      }
  1409→    }
  1410→
  1411→    final themeData = Theme.of(context);
  1412→
  1413→    // reset cached values before page renders
  1414→    _lastAyahIdx = -1;
  1415→    _lastRenderedWordIdx = -1;
  1416→
  1417→    if (Settings.instance.reflowMode) {
  1418→      return _reflowModeText(rowHeight, ayahData, themeData);
  1419→    }
  1420→
  1421→    List<Widget> widgets = [];
  1422→    const divider = Divider(color: Colors.grey, height: 1);
  1423→    final bigScreen = isBigScreen();
  1424→    final defaultTextStyle = _getQuranTextStyle(_textFontSize());
  1425→
  1426→    final mushaf = Settings.instance.mushaf;
  1427→    final isIndoPk = isIndoPak(mushaf);
  1428→    final fontSize = _textFontSize();
  1429→    final boxFit = bigScreen ? BoxFit.contain : BoxFit.scaleDown;
  1430→    final leftRightBorder = BoxDecoration(
  1431→      border: Border.symmetric(
  1432→        vertical: BorderSide(color: themeData.dividerColor, width: 1),
  1433→      ),
  1434→    );
  1435→
  1436→    for (final (idx, l) in widget._pageLines.indexed) {
  1437→      if (l.lineAyahs.first.ayahIndex < 0) {
  1438→        widgets.addAll(_buildSurahHeaddress(idx, l, rowHeight));
  1439→        continue;
  1440→      }
  1441→
  1442→      // 15 line uthmani is not separated by divider
  1443→      if (isIndoPk) widgets.add(divider);
  1444→      // except first line, as its the top border of page
  1445→      if (!isIndoPk && idx == 0) widgets.add(divider);
  1446→
  1447→      const double padding = 4;
  1448→      widgets.add(
  1449→        Container(
  1450→          height: rowHeight,
  1451→          width: double.infinity,
  1452→          padding: const EdgeInsets.only(left: padding, right: padding),
  1453→          decoration: leftRightBorder,
  1454→          child: FittedBox(
  1455→            fit: boxFit,
  1456→            child: _buildLine(
  1457→              l,
  1458→              idx,
  1459→              ayahData,
  1460→              fontSize,
  1461→              rowWidth - (2 * padding),
  1462→              defaultTextStyle,
  1463→              themeData,
  1464→              mushaf,
  1465→            ),
  1466→          ),
  1467→        ),
  1468→      );
  1469→    }
  1470→    widgets.add(divider);
  1471→
  1472→    return widgets;
  1473→  }

  1475→  @override
  1476→  Widget build(BuildContext context) {
  1477→    // The spans from the previous build are being replaced
  1478→    _disposeRecognizers();
  1479→
  1480→    final numPageLines = switch (Settings.instance.mushaf) {
  1481→      Mushaf.Indopak16Line => 16,
  1482→      Mushaf.Indopak15Line || Mushaf.Uthmani15Line => 15,
  1483→      Mushaf.Indopak13Line => 13,
  1484→    };
  1485→
  1486→    final double topBorderHeight = 16.0;
  1487→
  1488→    final double height =
  1489→        _availableHeight(context) -
  1490→        ( /*divider between lines(1px)*/ numPageLines +
  1491→            2 + // some extra space
  1492→            /*topborder=*/ topBorderHeight);
  1493→
  1494→    final double rowHeight = max((height / numPageLines), 10.0);
  1495→    final double rowWidth = MediaQuery.sizeOf(context).width;
  1496→    const double padding = 4;
  1497→    return Padding(
  1498→      padding: const EdgeInsets.only(left: padding, right: padding),
  1499→      child: ListView(
  1500→        physics: const ClampingScrollPhysics(),
  1501→        children: [
  1502→          // Top border
  1503→          _pageTopBorder(topBorderHeight),
  1504→          // Divider
  1505→          if (Settings.instance.reflowMode)
  1506→            const Divider(color: Colors.grey, height: 1),
  1507→          // The actual page text
  1508→          ..._pageLines(rowHeight, rowWidth - (2 * padding)),
  1509→        ],
  1510→      ),
  1511→    );
  1512→  }
