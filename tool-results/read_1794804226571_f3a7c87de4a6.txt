    64→const AR_DIGITS = ["٠", "١", "٢", "٣", "٤", "٥", "٦", "٧", "٨", "٩"]

    69→/** تطبيع كلمة للمقارنة (بلا تشكيل) */
    70→function plainWord(w: string): string {
    71→  return w
    72→    .replace(/[\u064B-\u065F\u0670\u06D6-\u06ED\u0640\u06DF\u200C\u200D]/g, "")
    73→    .replace(/\u0671/g, "\u0627")
    74→    .replace(/[؟،.,!؛:]/g, "")
    75→}

    77→/** فروق الكلمات بين نصين — يُبرز ما ليس مشتركاً في الطرف الآخر */
    78→function diffWords(focusText: string, otherText: string): { w: string; diff: boolean }[] {
    79→  const A = focusText.split(/\s+/).filter(Boolean)
    80→  const B = otherText.split(/\s+/).filter(Boolean)
    81→  const na = A.map(plainWord)
    82→  const nb = B.map(plainWord)
    83→  const m = na.length,
    84→    k = nb.length
    85→  if (m === 0 || k === 0) return B.map((w) => ({ w, diff: true }))
    86→  // LCS بمصفوفة كاملة للرجوع
    87→  const dp: Uint16Array[] = Array.from({ length: m + 1 }, () => new Uint16Array(k + 1))
    88→  for (let i = m - 1; i >= 0; i--) {
    89→    for (let j = k - 1; j >= 0; j--) {
    90→      dp[i][j] = na[i] === nb[j] ? dp[i + 1][j + 1] + 1 : Math.max(dp[i + 1][j], dp[i][j + 1])
    91→    }
    92→  }
    93→  const flags: boolean[] = new Array(k).fill(true)
    94→  let i = 0,
    95→    j = 0
    96→  const inLcs = new Set<number>()
    97→  while (i < m && j < k) {
    98→    if (na[i] === nb[j]) {
    99→      inLcs.add(j)
   100→      i++
   101→      j++
   102→    } else if (dp[i + 1][j] >= dp[i][j + 1]) {
   103→      i++
   104→    } else {
   105→      j++
   106→    }
   107→  }
   108→  for (let x = 0; x < k; x++) flags[x] = !inLcs.has(x) && nb[x] !== "" // علامات الوقف وحدها ليست فرقاً
   109→  return B.map((w, x) => ({ w, diff: flags[x] }))
   110→}

   112→/** تقسيم أعضاء مجموعة إلى مقاطع: آيات متتالية في نفس السورة = مقطع واحد */
   113→function passagesOf(items: AyahRef[]): AyahRef[][] {
   114→  const sorted = [...items].sort((a, b) => a.s - b.s || a.n - b.n)
   115→  const out: AyahRef[][] = []
   116→  for (const it of sorted) {
   117→    const last = out[out.length - 1]
   118→    if (last && last[0].s === it.s && last[last.length - 1].n === it.n - 1) last.push(it)
   119→    else out.push([it])
   120→  }
   121→  return out
   122→}

   133→      <span className="font-quran">{words}…</span>
   134→    </p>
   135→  )
