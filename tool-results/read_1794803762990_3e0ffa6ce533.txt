    12→  youngAdult, // 18-30: peak cognitive, can handle aggressive plans

    14→  middleAged, // 46-55: slower encoding, stronger discipline
    15→  senior, // 56-70: noticeably slower, shorter sessions
    16→  elderly, // 71+: fragile retention, very gentle pace

    79→  /// Computed age from birthday. Falls back to stored age for legacy profiles.
    80→  int get age {
    81→    if (birthday == null) return _ageFallback;
    82→    final now = DateTime.now();
    83→    int years = now.year - birthday!.year;
    84→    if (now.month < birthday!.month ||
    85→        (now.month == birthday!.month && now.day < birthday!.day)) {
    86→      years--;
    87→    }
    88→    return years.clamp(7, 100);
    89→  }
    90→
    91→  /// Static helper to calculate age from a birthday for use before profile creation.
    92→  static int calculateAge(DateTime birthday) {
    93→    final now = DateTime.now();
    94→    int years = now.year - birthday.year;
    95→    if (now.month < birthday.month ||
    96→        (now.month == birthday.month && now.day < birthday.day)) {
    97→      years--;
    98→    }
    99→    return years.clamp(0, 120);
   100→  }

   127→  /// Maps an actual age to the appropriate AgeGroup.
   128→  static AgeGroup ageGroupFromAge(int age) {
   129→    if (age <= 12) return AgeGroup.child;
   130→    if (age <= 17) return AgeGroup.teen;
   131→    if (age <= 30) return AgeGroup.youngAdult;
   132→    if (age <= 45) return AgeGroup.adult;
   133→    if (age <= 55) return AgeGroup.middleAged;
   134→    if (age <= 70) return AgeGroup.senior;
   135→    return AgeGroup.elderly;
   136→  }

   773→/// A snapshot of analytics data for a time period (week or month).
   774→class WeeklySnapshot {
   775→  final DateTime startDate;
   776→  final DateTime endDate;
   777→
   778→  // Session metrics
   779→  final int totalSessions;
   780→  final int totalDurationMinutes;
   781→  final double avgDurationMinutes;
   782→  final Map<int, int> sessionsPerDay; // day-of-week (1=Mon) → count
   783→
   784→  // Completion
   785→  final int plannedDays; // days with a plan
   786→  final int completedDays; // days plan was completed
   787→  final double completionRate; // 0.0–1.0
   788→
   789→  // Assessment distribution (across all sessions)
   790→  final int strongCount;
   791→  final int okayCount;
   792→  final int needsWorkCount;
   793→
   794→  // Progress
   795→  final int pagesMemorized; // new pages memorized this period
   796→  final int pagesReviewed; // pages reviewed this period
   797→  final double pagesPerWeek; // pace calculation
   798→
   799→  const WeeklySnapshot({
   800→    required this.startDate,
   801→    required this.endDate,
   802→    this.totalSessions = 0,
   803→    this.totalDurationMinutes = 0,
   804→    this.avgDurationMinutes = 0,
   805→    this.sessionsPerDay = const {},
   806→    this.plannedDays = 0,
   807→    this.completedDays = 0,
   808→    this.completionRate = 0,
   809→    this.strongCount = 0,
   810→    this.okayCount = 0,
   811→    this.needsWorkCount = 0,
   812→    this.pagesMemorized = 0,
   813→    this.pagesReviewed = 0,
   814→    this.pagesPerWeek = 0,
   815→  });
   816→
   817→  /// Total assessment count across all sessions.
   818→  int get totalAssessments => strongCount + okayCount + needsWorkCount;
