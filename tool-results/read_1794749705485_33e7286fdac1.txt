    26→    26→export function addDays(dateStr: string, days: number): string {
    27→    27→  const [y, m, d] = dateStr.split("-").map(Number)
    28→    28→  const dt = new Date(y, m - 1, d, 12)
    29→    29→  dt.setDate(dt.getDate() + days)
    30→    30→  return `${dt.getFullYear()}-${String(dt.getMonth() + 1).padStart(2, "0")}-${String(dt.getDate()).padStart(2, "0")}`
    31→    31→}

    33→    33→export function diffDays(a: string, b: string): number {
    34→    34→  const [ay, am, ad] = a.split("-").map(Number)
    35→    35→  const [by, bm, bd] = b.split("-").map(Number)
    36→    36→  const da = new Date(ay, am - 1, ad, 12)
    37→    37→  const db = new Date(by, bm - 1, bd, 12)
    38→    38→  return Math.round((da.getTime() - db.getTime()) / 86400000)
    39→    39→}

    87→    87→export function riskScore(
    88→    88→  p: { reviews: number; strength: number; misses: number; dueDate: string | null },
    89→    89→  today: string
    90→    90→): number {
    91→    91→  let risk = 0
    92→    92→  if (p.reviews === 0) risk += 9 // لم تُراجع بعد — أولوية قصوى
    93→    93→  if (p.dueDate) {
    94→    94→    const d = diffDays(today, p.dueDate)
    95→    95→    if (d > 0) risk += Math.min(d, 15) * 2 // متأخرة — كل يوم تأخير يرفع الخطر
    96→    96→    else if (d === 0) risk += 0.5 // مستحقة اليوم
    97→    97→    else risk -= Math.min(-d, 10) * 0.8 // مستقبلية — تنتظر دورها بهدوء
    98→    98→  } else {
    99→    99→    risk += 6 // محفوظة بلا موعد (احتياطي)
   100→   100→  }
   101→   101→  risk += (5 - p.strength) * 2.2 // الأضعف أخطر دائماً
   102→   102→  risk += Math.min(p.misses, 5) * 1.5 // سبق أن فاتتها مراجعات
   103→   103→  return risk
   104→   104→}

   112→   112→async function processPendingBefore(today: string) {
   113→   113→  const staleTasks = await db.task.findMany({
   114→   114→    where: {
   115→   115→      status: { in: ["pending", "snoozed"] },
   116→   116→      plan: { date: { lt: today } },
   117→   117→    },
   118→   118→    include: { plan: true },
   119→   119→  })
   120→   120→
   121→   121→  for (const task of staleTasks) {
   122→   122→    const gap = diffDays(today, task.plan.date)
   123→   123→    await db.task.update({ where: { id: task.id }, data: { status: "missed" } })
   124→   124→
   125→   125→    for (let p = task.pageFrom; p <= task.pageTo; p++) {
   126→   126→      const page = await db.pageState.findUnique({ where: { page: p } })
   127→   127→      if (!page || !page.memorized) continue
   128→   128→      // خصم لطيف: تنخفض القوة فقط إذا تأخرت كثيراً (3 أيام فأكثر)
   129→   129→      const decay = gap >= 3 ? Math.max(0, page.strength - 1) : page.strength
   130→   130→      await db.pageState.update({
   131→   131→        where: { page: p },
   132→   132→        data: {
   133→   133→          misses: { increment: 1 },
   134→   134→          strength: decay,
   135→   135→          dueDate: today, // مستحقة فوراً لتعود للخطة بأولوية
   136→   136→        },
   137→   137→      })
   138→   138→    }
   139→   139→
   140→   140→    const pages = task.pageTo - task.pageFrom + 1
   141→   141→    await db.activity
   142→   142→      .update({
   143→   143→        where: { date: task.plan.date },
   144→   144→        data: { missedPages: { increment: pages } },
   145→   145→      })
   146→   146→      .catch(() => {})
   147→   147→  }
   148→   148→
   149→   149→  // تحديث حالة الخطط القديمة
   150→   150→  const oldPlans = await db.dailyPlan.findMany({
   151→   151→    where: { date: { lt: today }, status: { in: ["open", "partial"] } },
   152→   152→    include: { tasks: true },
   153→   153→  })
   154→   154→  for (const plan of oldPlans) {
   155→   155→    const anyDone = plan.tasks.some((t) => t.status === "done")
   156→   156→    await db.dailyPlan.update({
   157→   157→      where: { id: plan.id },
   158→   158→      data: { status: anyDone ? "partial" : "missed" },
   159→   159→    })
   160→   160→  }
   161→   161→}
   162→   162→
   163→   163→// ==================== توليد خطة اليوم ====================
   164→   164→
   165→   165→/**
   166→   166→ * المخطط اليومي — مبدآن لا يُخرقان:
   167→   167→ * 1) وردك لا يتجاوز قدرتك أبداً (السعة الفعلية ≤ سعتك المحددة)
   168→   168→ * 2) يبدأ دائماً بالأعلى خطورة على النسيان: المتأخر ثم الضعيف ثم لم يُراجع أبداً
   169→   169→ * ومع كل مراجعة تُحدَّث فترة العودة بحيث لا تتجاوز قاعدة عدم النسيان (10 أيام افتراضياً).

   171→   171→export async function ensureTodayPlan(profile: {
   172→   172→  id: number
   173→   173→  timezone: string
   174→   174→  dailyCapacity: number
   175→   175→  unitSize: number
   176→   176→  intensity: string
   177→   177→  maxGapDays: number
   178→   178→  autoAdjust: boolean
   179→   179→}) {
   180→   180→  const today = todayStr(profile.timezone)
   181→   181→  const maxGap = effectiveMaxGap(profile.maxGapDays)
   182→   182→
   183→   183→  await processPendingBefore(today)
   184→   184→
   185→   185→  // قاعدة عدم النسيان: لا تنتظر أي صفحة محفوظة أكثر من maxGap يوماً
   186→   186→  // (يشمل ترحيل أي مواعيد قديمة كانت مجدولة بفترات أطول)
   187→   187→  const cutoff = addDays(today, maxGap)
   188→   188→  await db.pageState.updateMany({
   189→   189→    where: { memorized: true, dueDate: { gt: cutoff } },
   190→   190→    data: { dueDate: cutoff },
   191→   191→  })
   192→   192→
   193→   193→  const existing = await db.dailyPlan.findUnique({
   194→   194→    where: { date: today },
   195→   195→    include: { tasks: true },
   196→   196→  })
   197→   197→  if (existing) return existing
   198→   198→
   199→   199→  const pages = await db.pageState.findMany({ where: { memorized: true } })
   200→   200→  if (pages.length === 0) return null
   201→   201→
   202→   202→  // --- حساب السعة الفعلية (التكيف) ---
   203→   203→  let effCapacity = profile.dailyCapacity
   204→   204→  const weekAgo = addDays(today, -7)
   205→   205→  const recent = await db.activity.findMany({
   206→   206→    where: { date: { gte: weekAgo, lt: today } },
   207→   207→  })
   208→   208→  const recentPlans = recent.filter((a) => a.totalTasks > 0)
   209→   209→  if (profile.autoAdjust && recentPlans.length > 0) {
   210→   210→    const done = recentPlans.reduce((s, a) => s + a.donePages, 0)
   211→   211→    const missed = recentPlans.reduce((s, a) => s + a.missedPages, 0)
   212→   212→    const rate = done + missed > 0 ? done / (done + missed) : 1
   213→   213→    if (rate < 0.5) {
   214→   214→      effCapacity = Math.max(1, Math.round(profile.dailyCapacity * 0.7))
   215→   215→    } else if (rate < 0.8) {
   216→   216→      effCapacity = Math.max(1, Math.round(profile.dailyCapacity * 0.85))
   217→   217→    }
   218→   218→    // لا نرفع السعة فوق ما حددته أبداً — الورد يبقى دائماً في حدود قدرتك
   219→   219→  }
   220→   220→
   221→   221→  // --- اختيار بترتيب الخطورة: الأعلى خطراً على النسيان أولاً ---
   222→   222→  const due = pages.filter((p) => !p.dueDate || p.dueDate <= today)
   223→   223→  const dueSorted = [...due].sort(
   224→   224→    (a, b) => riskScore(b, today) - riskScore(a, today) || a.page - b.page
   225→   225→  )
   226→   226→  const picked: typeof dueSorted = dueSorted.slice(0, effCapacity)
   227→   227→
   228→   228→  // --- تعبئة لطيفة بالأقرب استحقاقاً حتى لا يفرغ وردك أبداً ---
   229→   229→  if (picked.length < effCapacity) {
   230→   230→    const upcoming = pages
   231→   231→      .filter((p) => p.dueDate && p.dueDate > today)
   232→   232→      .sort((a, b) => (a.dueDate! < b.dueDate! ? -1 : a.dueDate! > b.dueDate! ? 1 : 0))
   233→   233→    for (const p of upcoming) {
   234→   234→      if (picked.length >= effCapacity) break
   235→   235→      if (!picked.includes(p)) picked.push(p)
   236→   236→    }
   237→   237→  }
   238→   238→
   239→   239→  if (picked.length === 0) return null
   240→   240→
   241→   241→  // --- تجميع الصفحات المتجاورة في مهام ---
   242→   242→  const sorted = [...picked].sort((a, b) => a.page - b.page)
   243→   243→  const runs: { from: number; to: number }[] = []
   244→   244→  for (const p of sorted) {
   245→   245→    const last = runs[runs.length - 1]
   246→   246→    if (last && p.page === last.to + 1) last.to = p.page
   247→   247→    else runs.push({ from: p.page, to: p.page })
   248→   248→  }
   249→   249→  const unit = Math.max(1, profile.unitSize)
   250→   250→  const taskRanges: { from: number; to: number }[] = []
   251→   251→  for (const run of runs) {
   252→   252→    let cursor = run.from
   253→   253→    while (cursor <= run.to) {
   254→   254→      const to = Math.min(cursor + unit - 1, run.to)
   255→   255→      taskRanges.push({ from: cursor, to })
   256→   256→      cursor = to + 1
   257→   257→    }
   258→   258→  }
   259→   259→
   260→   260→  const plan = await db.dailyPlan.create({
   261→   261→    data: {
   262→   262→      date: today,
   263→   263→      capacity: effCapacity,
   264→   264→      status: "open",
   265→   265→      tasks: {
   266→   266→        create: taskRanges.map((r) => ({ pageFrom: r.from, pageTo: r.to })),
   267→   267→      },
   268→   268→    },
   269→   269→    include: { tasks: true },
   270→   270→  })
   271→   271→
   272→   272→  await db.activity.upsert({
   273→   273→    where: { date: today },
   274→   274→    update: { totalTasks: plan.tasks.length },
   275→   275→    create: { date: today, totalTasks: plan.tasks.length },
   276→   276→  })
   277→   277→
   278→   278→  return plan
   279→   279→}

   283→   283→export async function completeTask(taskId: number, quality: Quality) {
   284→   284→  const task = await db.task.findUnique({ where: { id: taskId }, include: { plan: true } })
   285→   285→  if (!task || task.status === "done") return { ok: false }
   286→   286→
   287→   287→  const profile = await getProfile()
   288→   288→  const today = todayStr(profile.timezone)
   289→   289→  const intervals = intervalsFor(profile.intensity)
   290→   290→  const maxGap = effectiveMaxGap(profile.maxGapDays)
   291→   291→
   292→   292→  const pagesCount = task.pageTo - task.pageFrom + 1
   293→   293→  let xpGain = 0
   294→   294→  const xpPer = quality === "perfect" ? 15 : quality === "good" ? 10 : 5
   295→   295→  xpGain = xpPer * pagesCount
   296→   296→
   297→   297→  for (let p = task.pageFrom; p <= task.pageTo; p++) {
   298→   298→    const page = await db.pageState.findUnique({ where: { page: p } })
   299→   299→    if (!page) continue
   300→   300→    const newStrength = applyStrength(page.strength, quality)
   301→   301→    // قاعدة عدم النسيان: مهما بلغ إتقانك لا تنتظر الصفحة أكثر من maxGap
   302→   302→    const interval = Math.min(intervals[newStrength], maxGap)
   303→   303→    await db.pageState.update({
   304→   304→      where: { page: p },
   305→   305→      data: {
   306→   306→        strength: newStrength,
   307→   307→        reviews: { increment: 1 },
   308→   308→        lastReviewedAt: new Date(),
   309→   309→        dueDate: addDays(today, interval),
   310→   310→      },
   311→   311→    })
   312→   312→  }
   313→   313→
   314→   314→  await db.task.update({
   315→   315→    where: { id: taskId },
   316→   316→    data: { status: "done", quality, completedAt: new Date() },
   317→   317→  })
   318→   318→
   319→   319→  // تحديث النشاط
   320→   320→  const activity = await db.activity.upsert({
   321→   321→    where: { date: today },
   322→   322→    update: {
   323→   323→      donePages: { increment: pagesCount },
   324→   324→      doneTasks: { increment: 1 },
   325→   325→      xpGained: { increment: xpGain },
   326→   326→    },
   327→   327→    create: { date: today, donePages: pagesCount, doneTasks: 1, xpGained: xpGain },
   328→   328→  })
   329→   329→
   330→   330→  // السلسلة اليومية
   331→   331→  let streak = profile.streak
   332→   332→  if (profile.lastActiveDate !== today) {
   333→   333→    streak = profile.lastActiveDate === addDays(today, -1) ? profile.streak + 1 : 1
   334→   334→  }
   335→   335→  const bestStreak = Math.max(profile.bestStreak, streak)
   336→   336→
   337→   337→  // مكافأة إتمام اليوم كاملاً
   338→   338→  const planTasks = await db.task.findMany({ where: { planId: task.planId } })
   339→   339→  const allDone = planTasks.every((t) => t.status === "done")
   340→   340→  if (allDone) {
   341→   341→    xpGain += 30
   342→   342→    await db.dailyPlan.update({ where: { id: task.planId }, data: { status: "completed" } })
   343→   343→  }
   344→   344→
   345→   345→  await db.profile.update({
   346→   346→    where: { id: 1 },
   347→   347→    data: {
   348→   348→      xp: { increment: xpGain },
   349→   349→      streak,
   350→   350→      bestStreak,
   351→   351→      lastActiveDate: today,
   352→   352→    },
   353→   353→  })
   354→   354→
   355→   355→  return { ok: true, allDone, xpGain, activityDone: activity.donePages }
   356→   356→}

   392→   392→  const tomorrow = addDays(today, 1)
   393→   393→
   394→   394→  const plan = await db.dailyPlan.findUnique({ where: { date: today }, include: { tasks: true } })

   403→   403→    const pages = task.pageTo - task.pageFrom + 1

   405→   405→    for (let p = task.pageFrom; p <= task.pageTo; p++) {
   406→   406→      await db.pageState
   407→   407→        .update({ where: { page: p }, data: { dueDate: tomorrow } })
   408→   408→        .catch(() => {})
   409→   409→    }
   410→   410→  }
   411→   411→
   412→   412→  const anyDone = plan.tasks.some((t) => t.status === "done")
   413→   413→  await db.dailyPlan.update({
   414→   414→    where: { id: plan.id },
   415→   415→    data: { status: anyDone ? "partial" : "missed" },
   416→   416→  })
   417→   417→
   418→   418→  if (missedPages > 0) {
   419→   419→    await db.activity.upsert({
   420→   420→      where: { date: today },
   421→   421→      update: { missedPages: { increment: missedPages } },
   422→   422→      create: { date: today, missedPages, totalTasks: plan.tasks.length },
   423→   423→    })
   424→   424→  }
   425→   425→
   426→   426→  return { ok: true }
   427→   427→}

   487→   487→export async function updateRanges(add: { from: number; to: number }[], remove: { from: number; to: number }[]) {
   488→   488→  const profile = await getProfile()
   489→   489→  const today = todayStr(profile.timezone)
   490→   490→
   491→   491→  for (const r of add) {
   492→   492→    for (let p = Math.max(1, r.from); p <= Math.min(TOTAL_PAGES, r.to); p++) {
   493→   493→      await db.pageState.upsert({
   494→   494→        where: { page: p },
   495→   495→        update: { memorized: true, dueDate: today },
   496→   496→        create: { page: p, memorized: true, strength: 0, dueDate: today },
   497→   497→      })
   498→   498→    }
   499→   499→  }
   500→   500→  for (const r of remove) {
   501→   501→    for (let p = Math.max(1, r.from); p <= Math.min(TOTAL_PAGES, r.to); p++) {
   502→   502→      await db.pageState.update({
   503→   503→        where: { page: p },
   504→   504→        data: { memorized: false, dueDate: null },
   505→   505→      }).catch(() => {})
   506→   506→    }
   507→   507→    // حذف المهام المعلقة التي تقع ضمن النطاق المحذوف
   508→   508→    const todayPlan = await db.dailyPlan.findUnique({ where: { date: today }, include: { tasks: true } })
   509→   509→    if (todayPlan) {
   510→   510→      for (const t of todayPlan.tasks) {
   511→   511→        if (t.status !== "done" && t.pageFrom >= r.from && t.pageTo <= r.to) {
   512→   512→          await db.task.delete({ where: { id: t.id } })
   513→   513→        }
   514→   514→      }
   515→   515→      const remaining = await db.task.findMany({ where: { planId: todayPlan.id } })
   516→   516→      if (remaining.length === 0) {
   517→   517→        await db.dailyPlan.delete({ where: { id: todayPlan.id } })
   518→   518→        await ensureTodayPlan(profile)
   519→   519→      }
   520→   520→    }
   521→   521→  }
   522→   522→  return { ok: true }
   523→   523→}

   537→   537→function dueLabelFor(dueDate: string | null, reviews: number, today: string): string {
   538→   538→  if (reviews === 0) return "أول مراجعة"
   539→   539→  if (!dueDate) return "مستحق اليوم"
   540→   540→  const d = diffDays(today, dueDate)
   541→   541→  if (d > 0) return `متأخر منذ ${d} ${d === 1 ? "يوم" : d === 2 ? "يومين" : d <= 10 ? "أيام" : "يوماً"}`
   542→   542→  if (d === 0) return "مستحق اليوم"
   543→   543→  return "مراجعة استباقية"
   544→   544→}
