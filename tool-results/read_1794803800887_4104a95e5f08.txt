    11→/** تاريخ اليوم بصيغة YYYY-MM-DD حسب المنطقة الزمنية */
    12→export function todayStr(tz: string): string {
    13→  try {
    14→    return new Intl.DateTimeFormat("en-CA", {
    15→      timeZone: tz,
    16→      year: "numeric",
    17→      month: "2-digit",
    18→      day: "2-digit",
    19→    }).format(new Date())
    20→  } catch {
    21→    const d = new Date()
    22→    return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`
    23→  }
    24→}

    26→export function addDays(dateStr: string, days: number): string {
    27→  const [y, m, d] = dateStr.split("-").map(Number)
    28→  const dt = new Date(y, m - 1, d, 12)
    29→  dt.setDate(dt.getDate() + days)
    30→  return `${dt.getFullYear()}-${String(dt.getMonth() + 1).padStart(2, "0")}-${String(dt.getDate()).padStart(2, "0")}`
    31→}

    33→export function diffDays(a: string, b: string): number {
    34→  const [ay, am, ad] = a.split("-").map(Number)
    35→  const [by, bm, bd] = b.split("-").map(Number)
    36→  const da = new Date(ay, am - 1, ad, 12)
    37→  const db = new Date(by, bm - 1, bd, 12)
    38→  return Math.round((da.getTime() - db.getTime()) / 86400000)
    39→}

    83→/**
    84→ * درجة خطورة النسيان لصفحة — كلما زادت زادت أولويتها في ورد اليوم.
    85→ * المحور الأساسي للخطة: التركيز الدائم على الضعيف والمتأخر وما لم يُراجع أبداً.
    86→ */
    87→export function riskScore(
    88→  p: { reviews: number; strength: number; misses: number; dueDate: string | null },
    89→  today: string
    90→): number {
    91→  let risk = 0
    92→  if (p.reviews === 0) risk += 9 // لم تُراجع بعد — أولوية قصوى
    93→  if (p.dueDate) {
    94→    const d = diffDays(today, p.dueDate)
    95→    if (d > 0) risk += Math.min(d, 15) * 2 // متأخرة — كل يوم تأخير يرفع الخطر
    96→    else if (d === 0) risk += 0.5 // مستحقة اليوم
    97→    else risk -= Math.min(-d, 10) * 0.8 // مستقبلية — تنتظر دورها بهدوء
    98→  } else {
    99→    risk += 6 // محفوظة بلا موعد (احتياطي)
   100→  }
   101→  risk += (5 - p.strength) * 2.2 // الأضعف أخطر دائماً
   102→  risk += Math.min(p.misses, 5) * 1.5 // سبق أن فاتتها مراجعات
   103→  return risk
   104→}

   108→/**

   112→async function processPendingBefore(today: string) {
   113→  const staleTasks = await db.task.findMany({
   114→    where: {
   115→      status: { in: ["pending", "snoozed"] },
   116→      plan: { date: { lt: today } },
   117→    },
   118→    include: { plan: true },
   119→  })
   120→
   121→  for (const task of staleTasks) {
   122→    const gap = diffDays(today, task.plan.date)
   123→    await db.task.update({ where: { id: task.id }, data: { status: "missed" } })
   124→
   125→    for (let p = task.pageFrom; p <= task.pageTo; p++) {
   126→      const page = await db.pageState.findUnique({ where: { page: p } })
   127→      if (!page || !page.memorized) continue
   128→      // خصم لطيف: تنخفض القوة فقط إذا تأخرت كثيراً (3 أيام فأكثر)
   129→      const decay = gap >= 3 ? Math.max(0, page.strength - 1) : page.strength
   130→      await db.pageState.update({
   131→        where: { page: p },
   132→        data: {
   133→          misses: { increment: 1 },
   134→          strength: decay,
   135→          dueDate: today, // مستحقة فوراً لتعود للخطة بأولوية
   136→        },
   137→      })
   138→    }
   139→
   140→    const pages = task.pageTo - task.pageFrom + 1
   141→    await db.activity
   142→      .update({
   143→        where: { date: task.plan.date },
   144→        data: { missedPages: { increment: pages } },
   145→      })
   146→      .catch(() => {})
   147→  }
   148→
   149→  // تحديث حالة الخطط القديمة
   150→  const oldPlans = await db.dailyPlan.findMany({
   151→    where: { date: { lt: today }, status: { in: ["open", "partial"] } },
   152→    include: { tasks: true },
   153→  })
   154→  for (const plan of oldPlans) {
   155→    const anyDone = plan.tasks.some((t) => t.status === "done")
   156→    await db.dailyPlan.update({
   157→      where: { id: plan.id },
   158→      data: { status: anyDone ? "partial" : "missed" },
   159→    })
   160→  }
   161→}
   162→
   163→// ==================== توليد خطة اليوم ====================
   164→
   165→/**
   166→ * المخطط اليومي — مبدآن لا يُخرقان:
   167→ * 1) وردك لا يتجاوز قدرتك أبداً (السعة الفعلية ≤ سعتك المحددة)
   168→ * 2) يبدأ دائماً بالأعلى خطورة على النسيان: المتأخر ثم الضعيف ثم لم يُراجع أبداً
   169→ * ومع كل مراجعة تُحدَّث فترة العودة بحيث لا تتجاوز قاعدة عدم النسيان (10 أيام افتراضياً).
   170→ */
   171→export async function ensureTodayPlan(profile: {
   172→  id: number
   173→  timezone: string
   174→  dailyCapacity: number
   175→  unitSize: number
   176→  intensity: string
   177→  maxGapDays: number
   178→  autoAdjust: boolean
   179→}) {
   180→  const today = todayStr(profile.timezone)
   181→  const maxGap = effectiveMaxGap(profile.maxGapDays)
   182→
   183→  await processPendingBefore(today)
   184→
   185→  // قاعدة عدم النسيان: لا تنتظر أي صفحة محفوظة أكثر من maxGap يوماً
   186→  // (يشمل ترحيل أي مواعيد قديمة كانت مجدولة بفترات أطول)
   187→  const cutoff = addDays(today, maxGap)
   188→  await db.pageState.updateMany({
   189→    where: { memorized: true, dueDate: { gt: cutoff } },
   190→    data: { dueDate: cutoff },
   191→  })
   192→
   193→  const existing = await db.dailyPlan.findUnique({
   194→    where: { date: today },
   195→    include: { tasks: true },
   196→  })
   197→  if (existing) return existing
   198→
   199→  const pages = await db.pageState.findMany({ where: { memorized: true } })
   200→  if (pages.length === 0) return null
   201→
   202→  // --- حساب السعة الفعلية (التكيف) ---
   203→  let effCapacity = profile.dailyCapacity
   204→  const weekAgo = addDays(today, -7)
   205→  const recent = await db.activity.findMany({
   206→    where: { date: { gte: weekAgo, lt: today } },
   207→  })
   208→  const recentPlans = recent.filter((a) => a.totalTasks > 0)
   209→  if (profile.autoAdjust && recentPlans.length > 0) {
   210→    const done = recentPlans.reduce((s, a) => s + a.donePages, 0)
   211→    const missed = recentPlans.reduce((s, a) => s + a.missedPages, 0)
   212→    const rate = done + missed > 0 ? done / (done + missed) : 1
   213→    if (rate < 0.5) {
   214→      effCapacity = Math.max(1, Math.round(profile.dailyCapacity * 0.7))
   215→    } else if (rate < 0.8) {
   216→      effCapacity = Math.max(1, Math.round(profile.dailyCapacity * 0.85))
   217→    }
   218→    // لا نرفع السعة فوق ما حددته أبداً — الورد يبقى دائماً في حدود قدرتك
   219→  }
   220→
   221→  // --- اختيار بترتيب الخطورة: الأعلى خطراً على النسيان أولاً ---
   222→  const due = pages.filter((p) => !p.dueDate || p.dueDate <= today)
   223→  const dueSorted = [...due].sort(
   224→    (a, b) => riskScore(b, today) - riskScore(a, today) || a.page - b.page
   225→  )
   226→  const picked: typeof dueSorted = dueSorted.slice(0, effCapacity)
   227→
   228→  // --- تعبئة لطيفة بالأقرب استحقاقاً حتى لا يفرغ وردك أبداً ---
   229→  if (picked.length < effCapacity) {
   230→    const upcoming = pages
   231→      .filter((p) => p.dueDate && p.dueDate > today)
   232→      .sort((a, b) => (a.dueDate! < b.dueDate! ? -1 : a.dueDate! > b.dueDate! ? 1 : 0))
   233→    for (const p of upcoming) {
   234→      if (picked.length >= effCapacity) break
   235→      if (!picked.includes(p)) picked.push(p)
   236→    }
   237→  }
   238→
   239→  if (picked.length === 0) return null
   240→
   241→  // --- تجميع الصفحات المتجاورة في مهام ---
   242→  const sorted = [...picked].sort((a, b) => a.page - b.page)
   243→  const runs: { from: number; to: number }[] = []
   244→  for (const p of sorted) {
   245→    const last = runs[runs.length - 1]
   246→    if (last && p.page === last.to + 1) last.to = p.page
   247→    else runs.push({ from: p.page, to: p.page })
   248→  }
   249→  const unit = Math.max(1, profile.unitSize)
   250→  const taskRanges: { from: number; to: number }[] = []
   251→  for (const run of runs) {
   252→    let cursor = run.from
   253→    while (cursor <= run.to) {
   254→      const to = Math.min(cursor + unit - 1, run.to)
   255→      taskRanges.push({ from: cursor, to })
   256→      cursor = to + 1
   257→    }
   258→  }
   259→
   260→  const plan = await db.dailyPlan.create({
   261→    data: {
   262→      date: today,
   263→      capacity: effCapacity,
   264→      status: "open",
   265→      tasks: {
   266→        create: taskRanges.map((r) => ({ pageFrom: r.from, pageTo: r.to })),
   267→      },
   268→    },
   269→    include: { tasks: true },
   270→  })
   271→
   272→  await db.activity.upsert({
   273→    where: { date: today },
   274→    update: { totalTasks: plan.tasks.length },
   275→    create: { date: today, totalTasks: plan.tasks.length },
   276→  })
   277→
   278→  return plan
   279→}

   283→export async function completeTask(taskId: number, quality: Quality) {
   284→  const task = await db.task.findUnique({ where: { id: taskId }, include: { plan: true } })
   285→  if (!task || task.status === "done") return { ok: false }
   286→
   287→  const profile = await getProfile()
   288→  const today = todayStr(profile.timezone)
   289→  const intervals = intervalsFor(profile.intensity)
   290→  const maxGap = effectiveMaxGap(profile.maxGapDays)
   291→
   292→  const pagesCount = task.pageTo - task.pageFrom + 1
   293→  let xpGain = 0
   294→  const xpPer = quality === "perfect" ? 15 : quality === "good" ? 10 : 5
   295→  xpGain = xpPer * pagesCount
   296→
   297→  for (let p = task.pageFrom; p <= task.pageTo; p++) {
   298→    const page = await db.pageState.findUnique({ where: { page: p } })
   299→    if (!page) continue
   300→    const newStrength = applyStrength(page.strength, quality)
   301→    // قاعدة عدم النسيان: مهما بلغ إتقانك لا تنتظر الصفحة أكثر من maxGap
   302→    const interval = Math.min(intervals[newStrength], maxGap)
   303→    await db.pageState.update({
   304→      where: { page: p },
   305→      data: {
   306→        strength: newStrength,
   307→        reviews: { increment: 1 },
   308→        lastReviewedAt: new Date(),
   309→        dueDate: addDays(today, interval),
   310→      },
   311→    })
   312→  }
   313→
   314→  await db.task.update({
   315→    where: { id: taskId },
   316→    data: { status: "done", quality, completedAt: new Date() },
   317→  })
   318→
   319→  // تحديث النشاط
   320→  const activity = await db.activity.upsert({
   321→    where: { date: today },
   322→    update: {
   323→      donePages: { increment: pagesCount },
   324→      doneTasks: { increment: 1 },
   325→      xpGained: { increment: xpGain },
   326→    },
   327→    create: { date: today, donePages: pagesCount, doneTasks: 1, xpGained: xpGain },
   328→  })
   329→
   330→  // السلسلة اليومية
   331→  let streak = profile.streak
   332→  if (profile.lastActiveDate !== today) {
   333→    streak = profile.lastActiveDate === addDays(today, -1) ? profile.streak + 1 : 1
   334→  }
   335→  const bestStreak = Math.max(profile.bestStreak, streak)
   336→
   337→  // مكافأة إتمام اليوم كاملاً
   338→  const planTasks = await db.task.findMany({ where: { planId: task.planId } })
   339→  const allDone = planTasks.every((t) => t.status === "done")
   340→  if (allDone) {
   341→    xpGain += 30
   342→    await db.dailyPlan.update({ where: { id: task.planId }, data: { status: "completed" } })
   343→  }
   344→
   345→  await db.profile.update({
   346→    where: { id: 1 },
   347→    data: {
   348→      xp: { increment: xpGain },
   349→      streak,
   350→      bestStreak,
   351→      lastActiveDate: today,
   352→    },
   353→  })
   354→
   355→  return { ok: true, allDone, xpGain, activityDone: activity.donePages }
   356→}

   431→export async function applyOnboarding(input: {
   432→  name: string
   433→  ranges: { from: number; to: number }[]
   434→  baseline: number
   435→  dailyCapacity: number
   436→  unitSize: number
   437→  intensity: Intensity
   438→  maxGapDays?: number
   439→  autoAdjust: boolean
   440→}) {
   441→  const today = todayStr("Africa/Cairo")
   442→  const maxGap = effectiveMaxGap(input.maxGapDays ?? 10)
   443→
   444→  // تهيئة كل الصفحات
   445→  const all: { page: number; memorized: boolean; strength: number; dueDate: string | null }[] = []
   446→  const mem = new Set<number>()
   447→  for (const r of input.ranges) {
   448→    for (let p = Math.max(1, r.from); p <= Math.min(TOTAL_PAGES, r.to); p++) mem.add(p)
   449→  }
   450→  for (let p = 1; p <= TOTAL_PAGES; p++) {
   451→    all.push({
   452→      page: p,
   453→      memorized: mem.has(p),
   454→      strength: mem.has(p) ? input.baseline : 0,
   455→      dueDate: mem.has(p) ? today : null,
   456→    })
   457→  }
   458→  await db.pageState.deleteMany({})
   459→  await db.pageState.createMany({ data: all })
   460→
   461→  await db.task.deleteMany({})
   462→  await db.dailyPlan.deleteMany({})
   463→  await db.activity.deleteMany({})
   464→
   465→  await db.profile.update({
   466→    where: { id: 1 },
   467→    data: {
   468→      onboarded: true,
   469→      name: input.name,
   470→      dailyCapacity: input.dailyCapacity,
   471→      unitSize: input.unitSize,
   472→      intensity: input.intensity,
   473→      maxGapDays: maxGap,
   474→      autoAdjust: input.autoAdjust,
   475→      xp: 0,
   476→      streak: 0,
   477→      bestStreak: 0,
   478→      lastActiveDate: null,
   479→    },
   480→  })
   481→
   482→  const profile = await getProfile()
   483→  await ensureTodayPlan(profile)
   484→  return { ok: true }
   485→}

   487→export async function updateRanges(add: { from: number; to: number }[], remove: { from: number; to: number }[]) {
   488→  const profile = await getProfile()
   489→  const today = todayStr(profile.timezone)
   490→
   491→  for (const r of add) {
   492→    for (let p = Math.max(1, r.from); p <= Math.min(TOTAL_PAGES, r.to); p++) {
   493→      await db.pageState.upsert({
   494→        where: { page: p },
   495→        update: { memorized: true, dueDate: today },
   496→        create: { page: p, memorized: true, strength: 0, dueDate: today },
   497→      })
   498→    }
   499→  }
   500→  for (const r of remove) {
   501→    for (let p = Math.max(1, r.from); p <= Math.min(TOTAL_PAGES, r.to); p++) {
   502→      await db.pageState.update({
   503→        where: { page: p },
   504→        data: { memorized: false, dueDate: null },
   505→      }).catch(() => {})
   506→    }
   507→    // حذف المهام المعلقة التي تقع ضمن النطاق المحذوف
   508→    const todayPlan = await db.dailyPlan.findUnique({ where: { date: today }, include: { tasks: true } })
   509→    if (todayPlan) {
   510→      for (const t of todayPlan.tasks) {
   511→        if (t.status !== "done" && t.pageFrom >= r.from && t.pageTo <= r.to) {
   512→          await db.task.delete({ where: { id: t.id } })
   513→        }
   514→      }
   515→      const remaining = await db.task.findMany({ where: { planId: todayPlan.id } })
   516→      if (remaining.length === 0) {
   517→        await db.dailyPlan.delete({ where: { id: todayPlan.id } })
   518→        await ensureTodayPlan(profile)
   519→      }
   520→    }
   521→  }
   522→  return { ok: true }
   523→}

   537→function dueLabelFor(dueDate: string | null, reviews: number, today: string): string {
   538→  if (reviews === 0) return "أول مراجعة"
   539→  if (!dueDate) return "مستحق اليوم"
   540→  const d = diffDays(today, dueDate)
   541→  if (d > 0) return `متأخر منذ ${d} ${d === 1 ? "يوم" : d === 2 ? "يومين" : d <= 10 ? "أيام" : "يوماً"}`
   542→  if (d === 0) return "مستحق اليوم"
   543→  return "مراجعة استباقية"
   544→}

   546→export async function buildAppState(): Promise<AppState> {
   547→  const profile = await getProfile()
   548→
   549→  if (!profile.onboarded) {
   550→    return {
   551→      today: todayStr(profile.timezone),
   552→      onboarded: false,
   553→      profile: {
   554→        name: "",
   555→        dailyCapacity: profile.dailyCapacity,
   556→        unitSize: profile.unitSize,
   557→        intensity: profile.intensity as Intensity,
   558→        autoAdjust: profile.autoAdjust,
   559→        timezone: profile.timezone,
   560→        xp: 0,
   561→        streak: 0,
   562→        bestStreak: 0,
   563→        maxGapDays: effectiveMaxGap(profile.maxGapDays),
   564→      },
   565→      level: { index: 0, name: "مبتدئ", nextThreshold: 150, progress: 0 },
   566→      plan: null,
   567→      overdueCount: 0,
   568→      forecast: [],
   569→      activity: [],
   570→      motivation: motivationFor(todayStr(profile.timezone)),
   571→      memorized: { total: 0, strengths: [] },
   572→      cycle: {
   573→        targetDays: effectiveMaxGap(profile.maxGapDays),
   574→        memorizedTotal: 0,
   575→        currentCycleDays: 0,
   576→        requiredDaily: 0,
   577→        coveragePct: 1,
   578→        coveredPages: 0,
   579→        onTrack: true,
   580→      },
   581→      attention: [],
   582→      stats: { totalDonePages: 0, completionRate: 0, remainingFirstPass: 0, cycleDays: 0, perfectDays: 0 },
   583→      badges: BADGES.map((b) => ({ id: b.id, earned: false })),
   584→    }
   585→  }
   586→
   587→  const today = todayStr(profile.timezone)
   588→
   589→  // توليد/جلب خطة اليوم (يعالج أيام الغياب تلقائياً)
   590→  const planRow = await ensureTodayPlan(profile)
   591→
   592→  // خريطة الصفحات
   593→  const pageRows = await db.pageState.findMany()
   594→  const strengths = new Array<number>(TOTAL_PAGES).fill(-1)
   595→  const memorizedRows = pageRows.filter((p) => p.memorized)
   596→  for (const p of memorizedRows) {
   597→    strengths[p.page - 1] = p.strength
   598→  }
   599→  const memorizedTotal = memorizedRows.length
   600→
   601→  // عدد المتأخرات (خارج خطة اليوم — ما لم يُجدول بعد)
   602→  const todayPlanPages = new Set<number>()
   603→  if (planRow) {
   604→    for (const t of planRow.tasks) {
   605→      for (let p = t.pageFrom; p <= t.pageTo; p++) todayPlanPages.add(p)
   606→    }
   607→  }
   608→  const overdueCount = pageRows.filter(
   609→    (p) => p.memorized && p.dueDate && p.dueDate < today && !todayPlanPages.has(p.page)
   610→  ).length
   611→
   612→  // ---- دورة الأمان (قاعدة عدم النسيان) ----
   613→  const maxGap = effectiveMaxGap(profile.maxGapDays)
   614→  const requiredDaily = memorizedTotal > 0 ? Math.ceil(memorizedTotal / maxGap) : 0
   615→  const currentCycleDays =
   616→    memorizedTotal > 0 && profile.dailyCapacity > 0 ? Math.ceil(memorizedTotal / profile.dailyCapacity) : 0
   617→  const windowStart = addDays(today, -(maxGap - 1))
   618→  const coveredPages = memorizedRows.filter(
   619→    (p) => p.reviews > 0 && p.lastReviewedAt && dateOnlyInTz(p.lastReviewedAt, profile.timezone) >= windowStart
   620→  ).length
   621→  const coveragePct = memorizedTotal > 0 ? coveredPages / memorizedTotal : 1
   622→
   623→  // ---- الصفحات الأكثر احتياجاً للمراجعة (خارج ورد اليوم) ----
   624→  const attention: AttentionPage[] = memorizedRows
   625→    .filter((p) => !todayPlanPages.has(p.page))
   626→    .map((p) => {
   627→      const risk = riskScore(p, today)
   628→      const od = p.dueDate ? Math.max(0, diffDays(today, p.dueDate)) : 0
   629→      let reason = ""
   630→      let kind: AttentionPage["kind"] = "weak"
   631→      if (p.reviews === 0) {
   632→        reason = "لم تُراجعها بعد"
   633→        kind = "never"
   634→      } else if (od >= 1) {
   635→        reason = `متأخرة ${formatDays(od)}`
   636→        kind = "overdue"
   637→      } else if (p.strength <= 1) {
   638→        reason = "ضعيفة — تحتاج تثبيتاً"
   639→        kind = "weak"
   640→      } else if (p.misses >= 2) {
   641→        reason = `فاتتها ${p.misses} ${p.misses === 2 ? "مرتين" : "مرات"} من قبل`
   642→        kind = "misses"
   643→      } else {
   644→        reason = "مستحقة قريباً — راقبها"
   645→        kind = "weak"
   646→      }
   647→      return {
   648→        page: p.page,
   649→        juz: pageToJuz(p.page),
   650→        label: pagesLabel(p.page, p.page),
   651→        reason,
   652→        kind,
   653→        overdueDays: od,
   654→        strength: p.strength,
   655→        risk,
   656→      }
   657→    })
   658→    .filter((a) => a.risk >= 3)
   659→    .sort((a, b) => b.risk - a.risk || a.page - b.page)
   660→    .slice(0, 10)
   661→
   662→  // توقعات الأسبوع
   663→  const forecast: { date: string; count: number }[] = []
   664→  for (let d = 1; d <= 7; d++) {
   665→    const date = addDays(today, d)
   666→    const count = pageRows.filter((p) => p.memorized && p.dueDate === date).length
   667→    forecast.push({ date, count })
   668→  }
   669→
   670→  // النشاط (آخر 120 يوماً)
   671→  const since = addDays(today, -120)
   672→  const activityRows = await db.activity.findMany({
   673→    where: { date: { gte: since } },
   674→    orderBy: { date: "asc" },
   675→  })
   676→
   677→  // إحصاءات
   678→  const totalDonePages = activityRows.reduce((s, a) => s + a.donePages, 0)
   679→  const last30 = activityRows.filter((a) => a.date >= addDays(today, -30) && a.totalTasks > 0)
   680→  const completionRate =
   681→    last30.length > 0
   682→      ? last30.reduce((s, a) => s + a.doneTasks / a.totalTasks, 0) / last30.length
   683→      : 0
   684→  const remainingFirstPass = pageRows.filter((p) => p.memorized && p.reviews === 0).length
   685→  const cycleDays = currentCycleDays // الدورة الكاملة الفعلية بسعتك الحالية
   686→  const perfectDays = activityRows.filter((a) => a.totalTasks > 0 && a.doneTasks === a.totalTasks).length
   687→
   688→  // الأوسمة
   689→  const allActivity = await db.activity.findMany({ orderBy: { date: "asc" } })
   690→  const totalDoneAll = allActivity.reduce((s, a) => s + a.donePages, 0)
   691→  const fullCoverage =
   692→    memorizedTotal > 0 && pageRows.filter((p) => p.memorized).every((p) => p.reviews > 0)
   693→  let comeback = false
   694→  const activeDates = allActivity.filter((a) => a.doneTasks > 0).map((a) => a.date)
   695→  for (let i = 1; i < activeDates.length; i++) {
   696→    if (diffDays(activeDates[i], activeDates[i - 1]) >= 3) comeback = true
   697→  }
   698→  const earnedMap: Record<string, boolean> = {
   699→    first_step: totalDoneAll >= 1,
   700→    perfect_day: perfectDays >= 1,
   701→    streak_3: profile.bestStreak >= 3,
   702→    streak_7: profile.bestStreak >= 7,
   703→    streak_14: profile.bestStreak >= 14,
   704→    streak_30: profile.bestStreak >= 30,
   705→    pages_50: totalDoneAll >= 50,
   706→    pages_100: totalDoneAll >= 100,
   707→    pages_500: totalDoneAll >= 500,
   708→    pages_1000: totalDoneAll >= 1000,
   709→    full_coverage: fullCoverage,
   710→    comeback,
   711→  }
   712→  const badges = BADGES.map((b) => ({ id: b.id, earned: earnedMap[b.id] ?? false }))
   713→
   714→  // بناء خطة اليوم بصيغة DTO
   715→  let plan: PlanDTO | null = null
   716→  if (planRow) {
   717→    const taskDTOs: TaskDTO[] = planRow.tasks
   718→      .map((t) => {
   719→        const pagesInRange: typeof pageRows = []
   720→        for (let p = t.pageFrom; p <= t.pageTo; p++) {
   721→          const row = pageRows.find((x) => x.page === p)
   722→          if (row) pagesInRange.push(row)
   723→        }
   724→        const minDue = pagesInRange.reduce<string | null>((m, p) => {
   725→          if (!p.dueDate) return m
   726→          return !m || p.dueDate < m ? p.dueDate : m
   727→        }, null)
   728→        const anyNew = pagesInRange.some((p) => p.reviews === 0)
   729→        return {
   730→          id: t.id,
   731→          pageFrom: t.pageFrom,
   732→          pageTo: t.pageTo,
   733→          pages: t.pageTo - t.pageFrom + 1,
   734→          juz: pageToJuz(t.pageFrom),
   735→          surahs: surahsInRange(t.pageFrom, t.pageTo),
   736→          label: pagesLabel(t.pageFrom, t.pageTo),
   737→          status: t.status as TaskDTO["status"],
   738→          quality: (t.quality as Quality | null) ?? null,
   739→          dueDate: minDue,
   740→          dueLabel: dueLabelFor(minDue, anyNew ? 0 : 1, today),
   741→          isNew: anyNew,
   742→        }
   743→      })
   744→      .sort((a, b) => a.pageFrom - b.pageFrom)
   745→
   746→    plan = {
   747→      date: planRow.date,
   748→      status: planRow.status as PlanDTO["status"],
   749→      capacity: planRow.capacity,
   750→      tasks: taskDTOs,
   751→    }
   752→  }
   753→
   754→  const level = levelFromXp(profile.xp)
   755→
   756→  return {
   757→    today,
   758→    onboarded: true,
   759→    profile: {
   760→      name: profile.name,
   761→      dailyCapacity: profile.dailyCapacity,
   762→      unitSize: profile.unitSize,
   763→      intensity: profile.intensity as Intensity,
   764→      autoAdjust: profile.autoAdjust,
   765→      timezone: profile.timezone,
   766→      xp: profile.xp,
   767→      streak: profile.streak,
   768→      bestStreak: profile.bestStreak,
   769→      maxGapDays: maxGap,
   770→    },
   771→    level,
   772→    plan,
   773→    overdueCount,
   774→    forecast,
   775→    activity: activityRows.map((a) => ({
   776→      date: a.date,
   777→      donePages: a.donePages,
   778→      missedPages: a.missedPages,
   779→      doneTasks: a.doneTasks,
   780→      totalTasks: a.totalTasks,
   781→    })),
   782→    motivation: motivationFor(today),
   783→    memorized: { total: memorizedTotal, strengths },
   784→    cycle: {
   785→      targetDays: maxGap,
   786→      memorizedTotal,
   787→      currentCycleDays,
   788→      requiredDaily,
   789→      coveragePct,
   790→      coveredPages,
   791→      onTrack: currentCycleDays <= maxGap,
   792→    },
   793→    attention,
   794→    stats: {
   795→      totalDonePages,
   796→      completionRate,
   797→      remainingFirstPass,
   798→      cycleDays,
   799→      perfectDays,
   800→    },
   801→    badges,

   803→}
