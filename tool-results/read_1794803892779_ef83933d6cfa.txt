    99→export function partialRatio(short: string, long: string): number {
   100→    if (!short || !long)
   101→        return 0.0;
   102→    if (short.length > long.length)
   103→        [short, long] = [long, short];
   104→    const window = short.length;
   105→    let best = 0.0;
   106→    for (let i = 0; i <= Math.max(0, long.length - window); i++) {
   107→        const r = ratio(short, long.slice(i, i + window));
   108→        if (r > best) {
   109→            best = r;
   110→            if (best === 1.0)
   111→                break;
   112→        }
   113→    }
   114→    return best;
   115→}

   239→    search(text: string, topK = 5): (QuranVerse & {
   240→        score: number;
   241→    })[] {
   242→        const scored: (QuranVerse & { score: number })[] = [];
   243→        for (const v of this.verses) {
   244→            const score = ratio(text, v.phonemes_joined);
   245→            scored.push({ ...v, score });
   246→        }
   247→        scored.sort((a, b) => b.score - a.score);

   250→    retrieveCandidates(text: string, { maxSpan = 4, hint = null, singleLimit = 32, topSurahs = 3, spanLimit = 32, }: RetrievalOptions = {}): CandidateRetrieval {
   251→        if (!text.trim()) {
   252→            return { singles: [], spans: [], combined: [] };
   253→        }
   254→        const bonuses = this._continuationBonuses(hint);
   255→        const textWords = text.split(/\s+/).filter(Boolean);
   256→        const noSpaceText = text.replace(/ /g, "");
   257→        const scored: [QuranVerse, number, number, number][] = [];
   258→        for (const v of this.verses) {
   259→            let raw = ratio(text, v.phonemes_joined);
   260→            const verseWords = v.phoneme_words;
   261→            const sharedWordCount = Math.min(textWords.length, verseWords.length);
   262→            if (sharedWordCount > 0) {
   263→                const textPrefix = textWords.slice(0, sharedWordCount).join(" ");
   264→                const versePrefix = verseWords.slice(0, sharedWordCount).join(" ");
   265→                raw = Math.max(raw, ratio(textPrefix, versePrefix));
   266→            }
   267→            if (noSpaceText.length <= 10) {
   268→                raw = Math.max(raw, this._shortQueryBoost(noSpaceText, v));
   269→            }
   270→            if (v.phonemes_joined_no_bsm) {
   271→                raw = Math.max(raw, ratio(text, v.phonemes_joined_no_bsm));
   272→                if (noSpaceText.length <= 10) {
   273→                    raw = Math.max(raw, this._shortQueryBoost(noSpaceText, v, true));
   274→                }
   275→            }
   276→            const bonus = bonuses.get(`${v.surah}:${v.ayah}`) ?? 0.0;
   277→            if (bonus > 0) {
   278→                const sp = QuranDB._suffixPrefixScore(text, v.phonemes_joined);
   279→                raw = Math.max(raw, sp);
   280→            }
   281→            scored.push([v, raw, bonus, Math.min(raw + bonus, 1.0)]);
   282→        }
   283→        scored.sort((a, b) => b[3] - a[3]);
   284→        const pass2Surahs: number[] = [];
   285→        for (let i = 0; i < scored.length && pass2Surahs.length < topSurahs; i++) {
   286→            const surah = scored[i][0].surah;
   287→            if (!pass2Surahs.includes(surah)) {
   288→                pass2Surahs.push(surah);
   289→            }
   290→        }
   291→        if (noSpaceText.length >= 8) {
   292→            let resorted = false;
   293→            for (let i = 0; i < scored.length; i++) {
   294→                const [v, raw, bonus] = scored[i];
   295→                if (noSpaceText.length >= (v.phonemes_joined_ns?.length ?? 0) * 0.8)
   296→                    continue;
   297→                let frag = fragmentScore(noSpaceText, v.phonemes_joined_ns ?? "");
   298→                if (v.phonemes_joined_no_bsm_ns) {
   299→                    frag = Math.max(frag, fragmentScore(noSpaceText, v.phonemes_joined_no_bsm_ns));
   300→                }
   301→                if (frag > raw) {
   302→                    const boosted = raw + (frag - raw) * 0.7;
   303→                    scored[i] = [v, boosted, bonus, Math.min(boosted + bonus, 1.0)];
   304→                    resorted = true;
   305→                }
   306→            }
   307→            if (resorted)
   308→                scored.sort((a, b) => b[3] - a[3]);
   309→        }
   310→        const singles = scored
   311→            .slice(0, singleLimit)
   312→            .map(([v, raw, bonus, total]) => this._candidateFromVerse(v, raw, bonus, total));
   313→        const spans: QuranCandidate[] = [];
   314→        for (let surahRank = 0; surahRank < pass2Surahs.length; surahRank++) {
   315→            const surah = pass2Surahs[surahRank];
   316→            const verses = this._bySurah.get(surah) ?? [];
   317→            for (let i = 0; i < verses.length; i++) {
   318→                for (let span = 2; span <= maxSpan; span++) {
   319→                    if (i + span > verses.length)
   320→                        break;
   321→                    const chunk = verses.slice(i, i + span);
   322→                    const spanText = this._joinedSpanPhonemes(chunk);
   323→                    let raw = ratio(text, spanText);
   324→                    const spanWords = spanText.split(/\s+/).filter(Boolean);
   325→                    const sharedWordCount = Math.min(textWords.length, spanWords.length);
   326→                    if (sharedWordCount > 0) {
   327→                        const textPrefix = textWords.slice(0, sharedWordCount).join(" ");
   328→                        const spanPrefix = spanWords.slice(0, sharedWordCount).join(" ");
   329→                        raw = Math.max(raw, ratio(textPrefix, spanPrefix));
   330→                    }
   331→                    const bonus = bonuses.get(`${chunk[0].surah}:${chunk[0].ayah}`) ?? 0.0;
   332→                    const score = Math.min(raw + bonus, 1.0);
   333→                    spans.push(this._candidateFromSpan(chunk, raw, bonus, score, surahRank));
   334→                }
   335→            }
   336→        }
   337→        spans.sort((a, b) => b.stage_a_score - a.stage_a_score);
   338→        return {
   339→            singles,
   340→            spans: spans.slice(0, spanLimit),
   341→            combined: singles.concat(spans.slice(0, spanLimit)),
   342→        };
   343→    }
   344→    matchVerse(text: string, threshold = 0.3, maxSpan = 3, hint: [number, number] | null = null, returnTopK = 0): QuranChampionMatch | null {
   345→        const retrieved = this.retrieveCandidates(text, {
   346→            maxSpan,
   347→            hint,
   348→            singleLimit: Math.max(returnTopK, 5),
   349→            topSurahs: 20,
   350→            spanLimit: 64,
   351→        });
   352→        const ranked = retrieved.combined
   353→            .slice()
   354→            .sort((a, b) => b.stage_a_score - a.stage_a_score);
   355→        const best = ranked[0];
   356→        if (!best || best.stage_a_score < threshold) {
   357→            return null;
   358→        }
   359→        const result: QuranChampionMatch = {
   360→            surah: best.surah,
   361→            ayah: best.ayah,
   362→            ayah_end: best.ayah_end,
   363→            text: best.text,
   364→            phonemes_joined: best.phonemes_joined,
   365→            score: best.stage_a_score,
   366→            raw_score: best.raw_score,
   367→            bonus: best.bonus,
   368→        };
   369→        if (returnTopK > 0) {
   370→            result.runners_up = retrieved.singles.slice(0, returnTopK).map((candidate) => ({
   371→                surah: candidate.surah,
   372→                ayah: candidate.ayah,
   373→                raw_score: Math.round(candidate.raw_score * 1000) / 1000,
   374→                bonus: Math.round(candidate.bonus * 1000) / 1000,
   375→                score: Math.round(candidate.stage_a_score * 1000) / 1000,
   376→                phonemes_joined: candidate.phonemes_joined.slice(0, 60),
   377→            }));
   378→        }
   379→        return result;
   380→    }

   384→    bestJoint03Match(text: string): QuranChampionMatch | null {
   385→        const top = this._joint02MatchPhonemeText(text, JOINT_TOP_K_LEVENSHTEIN);
   386→        if (!top.length)
   387→            return null;
   388→        const best = top[0];
   389→        const bestScore = best.score;
   390→        // Selection logic below is unchanged; it only assigns `winner` at the
   391→        // former return points so a single exit can annotate confidence.
   392→        let winner: QuranChampionMatch | undefined;
   393→        const shortOpening = this._jointShortOpeningSpanCandidate(text, bestScore);
   394→        if (shortOpening) {
   395→            winner = shortOpening;
   396→        }
   397→        else {
   398→            const bestIsLateSpan = best.ayah_end != null && best.ayah > 1;
   399→            const lowConfidence = bestScore < 0.62;
   400→            if (!bestIsLateSpan && !lowConfidence) {
   401→                winner = best;
   402→            }
   403→            else {
   404→                const noSpaceLen = text.replace(/ /g, "").length;
   405→                const prefix = this._jointSurahPrefixCandidates(text);
   406→                const globalSpan = this._jointGlobalSpanCandidates(text);
   407→                const candidates = [best]
   408→                    .concat(prefix.filter((p) => p.score >= bestScore + JOINT_PREFIX_MARGIN))
   409→                    .concat(globalSpan.filter((g) => g.score >= bestScore + JOINT_GLOBAL_SPAN_MARGIN));
   410→                candidates.sort((a, b) => b.score - a.score);
   411→                const chosen = candidates[0];
   412→                if (noSpaceLen >= JOINT_OPENING_COLLAPSE_MIN_CHARS &&
   413→                    noSpaceLen <= JOINT_OPENING_COLLAPSE_MAX_CHARS &&
   414→                    best.ayah_end != null &&
   415→                    best.ayah > 1) {
   416→                    const sameSurahPrefix = prefix
   417→                        .filter((p) => p.surah === best.surah &&
   418→                        p.score >= JOINT_OPENING_COLLAPSE_MIN_SCORE &&
   419→                        (p.ayah_end == null || best.ayah_end == null || p.ayah_end >= best.ayah_end))
   420→                        .sort((a, b) => b.score - a.score);
   421→                    if (sameSurahPrefix.length > 0)
   422→                        winner = sameSurahPrefix[0];
   423→                }
   424→                if (!winner)
   425→                    winner = chosen;
   426→            }
   427→        }
   428→        this._annotateConfidence(winner, top, text);
   429→        return winner;
   430→    }
   431→    /**
   432→     * Attach confidence signals to a chosen match without affecting selection:
   433→     *  - `margin`: score() gap to the best DISTINCT verse in the ranked list.
   434→     *    A large margin means the match is unambiguous even when its absolute
   435→     *    score is depressed (e.g. a short fragment of a long verse).
   436→     *  - `frag_fit`: pure semi-global fragment fit of the query against the
   437→     *    match's phonemes (1.0 = query is an exact substring of the match).
   438→     * These let the trust gate accept distinctive fragments that the flat
   439→     * absolute-score threshold alone would wrongly reject.
   440→     */
   441→    private _annotateConfidence(winner: QuranChampionMatch, top: QuranChampionMatch[], text: string): void {
   442→        const noSpaceText = text.replace(/ /g, "");
   443→        const refNs = (winner.phonemes_joined ?? "").replace(/ /g, "");
   444→        const winnerFit = refNs && noSpaceText ? fragmentScore(noSpaceText, refNs) : 0;
   445→        winner.frag_fit = QuranDB._round4(winnerFit);
   446→        // Best score and best fragment-fit among TRULY DIFFERENT candidates (not
   447→        // the winner and not a same-surah span that overlaps it — an overlapping
   448→        // span is the same answer, not a rival). Score-margin is unreliable for
   449→        // fragments: a short verse with coincidental whole-string overlap scores
   450→        // high but fits poorly. Fit-margin is the decisive discriminator.
   451→        let bestOtherScore = 0;
   452→        let bestOtherFit = 0;
   453→        for (const c of top) {
   454→            if (this._sameOrOverlapping(c, winner))
   455→                continue;
   456→            if (c.score > bestOtherScore)
   457→                bestOtherScore = c.score;
   458→            const cns = (c.phonemes_joined ?? "").replace(/ /g, "");
   459→            if (cns && noSpaceText) {
   460→                const f = fragmentScore(noSpaceText, cns);
   461→                if (f > bestOtherFit)
   462→                    bestOtherFit = f;
   463→            }
   464→        }
   465→        winner.margin = QuranDB._round4(winner.score - bestOtherScore);
   466→        winner.fit_margin = QuranDB._round4(winnerFit - bestOtherFit);
   467→    }
   468→    /** True when candidate c is the winner or a same-surah span that overlaps it. */
   469→    private _sameOrOverlapping(c: QuranChampionMatch, w: QuranChampionMatch): boolean {
   470→        if (c.surah !== w.surah)
   471→            return false;
   472→        const cEnd = c.ayah_end ?? c.ayah;
   473→        const wEnd = w.ayah_end ?? w.ayah;
   474→        return c.ayah <= wEnd && cEnd >= w.ayah;
   475→    }

   491→    private _joint02MatchPhonemeText(phonemeText: string, topK = JOINT_TOP_K_LEVENSHTEIN): QuranChampionMatch[] {
   492→        if (!phonemeText.trim())
   493→            return [];
   494→        const noSpaceText = phonemeText.replace(/ /g, "");
   495→        const scored: [QuranVerse, number, number][] = [];
   496→        for (const verse of this._jointCandidateVerses(noSpaceText)) {
   497→            const ref = verse.phonemes_joined;
   498→            if (!ref)
   499→                continue;
   500→            let raw = ratio(phonemeText, ref);
   501→            if (noSpaceText.length <= 10) {
   502→                raw = Math.max(raw, this._shortQueryBoost(noSpaceText, verse));
   503→            }
   504→            const noBsm = verse.phonemes_joined_no_bsm;
   505→            if (noBsm) {
   506→                raw = Math.max(raw, ratio(phonemeText, noBsm));
   507→                if (noSpaceText.length <= 10) {
   508→                    raw = Math.max(raw, this._shortQueryBoost(noSpaceText, verse, true));
   509→                }
   510→            }
   511→            scored.push([verse, raw, raw]);
   512→        }
   513→        scored.sort((a, b) => b[2] - a[2]);
   514→        const pass2Surahs: number[] = [];
   515→        for (const [verse] of scored) {
   516→            if (!pass2Surahs.includes(verse.surah)) {
   517→                pass2Surahs.push(verse.surah);
   518→            }
   519→            if (pass2Surahs.length >= JOINT_TOP_SURAHS)
   520→                break;
   521→        }
   522→        if (noSpaceText.length >= 8) {
   523→            let resorted = false;
   524→            for (let i = 0; i < scored.length; i++) {
   525→                const [verse, raw] = scored[i];
   526→                const refNs = verse.phonemes_joined_ns ?? "";
   527→                if (!refNs || noSpaceText.length >= refNs.length * 0.8)
   528→                    continue;
   529→                let frag = fragmentScore(noSpaceText, refNs);
   530→                const noBsmNs = verse.phonemes_joined_no_bsm_ns;
   531→                if (noBsmNs) {
   532→                    frag = Math.max(frag, fragmentScore(noSpaceText, noBsmNs));
   533→                }
   534→                if (frag > raw) {
   535→                    const boosted = raw + (frag - raw) * JOINT_FRAGMENT_BLEND;
   536→                    scored[i] = [verse, boosted, boosted];
   537→                    resorted = true;
   538→                }
   539→            }
   540→            if (resorted)
   541→                scored.sort((a, b) => b[2] - a[2]);
   542→        }
   543→        const spanResults: QuranChampionMatch[] = [];
   544→        for (const surahNum of pass2Surahs) {
   545→            const verses = this._bySurah.get(surahNum) ?? [];
   546→            for (let i = 0; i < verses.length; i++) {
   547→                for (let span = 2; span <= JOINT_MAX_SPAN; span++) {
   548→                    if (i + span > verses.length)
   549→                        break;
   550→                    const chunk = verses.slice(i, i + span);
   551→                    const spanPhonemes = this._joinedSpanPhonemes(chunk);
   552→                    const score = QuranDB._round4(ratio(phonemeText, spanPhonemes));
   553→                    spanResults.push({
   554→                        surah: surahNum,
   555→                        ayah: chunk[0].ayah,
   556→                        ayah_end: chunk[chunk.length - 1].ayah,
   557→                        text: chunk.map((verse) => verse.text_uthmani).join(" "),
   558→                        phonemes_joined: spanPhonemes,
   559→                        score,
   560→                        raw_score: score,
   561→                        bonus: 0,
   562→                    });
   563→                }
   564→            }
   565→        }
   566→        const singles = scored.slice(0, Math.max(topK, 32)).map(([verse, raw, boosted]): QuranChampionMatch => {
   567→            const score = QuranDB._round4(boosted);
   568→            return {
   569→                surah: verse.surah,
   570→                ayah: verse.ayah,
   571→                ayah_end: null,
   572→                text: verse.text_uthmani,
   573→                phonemes_joined: verse.phonemes_joined,
   574→                score,
   575→                raw_score: QuranDB._round4(raw),
   576→                bonus: 0,
   577→            };
   578→        });
   579→        return singles
   580→            .concat(spanResults)
   581→            .sort((a, b) => b.score - a.score)
   582→            .slice(0, topK);
   583→    }

   644→    private _jointShortOpeningSpanCandidate(phonemeText: string, bestScore: number): QuranChampionMatch | null {
   645→        const noSpaceText = phonemeText.replace(/ /g, "");
   646→        if (noSpaceText.length < JOINT_SHORT_OPENING_MIN_QUERY_CHARS)
   647→            return null;
   648→        let best: QuranChampionMatch | null = null;
   649→        for (const row of this._jointPrefixSpanTable()) {
   650→            const first = this.getVerse(row.surah, 1);
   651→            const second = this.getVerse(row.surah, 2);
   652→            if (!first || !second)
   653→                continue;
   654→            const firstNs = first.phonemes_joined_no_bsm_ns ??
   655→                first.phonemes_joined_ns ??
   656→                first.phonemes_joined.replace(/ /g, "");
   657→            if (!firstNs || firstNs.length > JOINT_SHORT_OPENING_MAX_CHARS)
   658→                continue;
   659→            if (!noSpaceText.startsWith(firstNs))
   660→                continue;
   661→            const remainder = noSpaceText.slice(firstNs.length);
   662→            const secondNs = second.phonemes_joined_ns ?? second.phonemes_joined.replace(/ /g, "");
   663→            const compareLen = Math.min(secondNs.length, remainder.length, 12);
   664→            if (compareLen >= 4 &&
   665→                ratio(remainder.slice(0, compareLen), secondNs.slice(0, compareLen)) < 0.72) {
   666→                continue;
   667→            }
   668→            const raw = ratio(phonemeText, row.phonemes_joined);
   669→            const frag = fragmentScore(noSpaceText, row.phonemes_joined.replace(/ /g, ""));
   670→            const score = Math.max(raw, raw + (frag - raw) * JOINT_FRAGMENT_BLEND);
   671→            if (score >= JOINT_SHORT_OPENING_MIN_SCORE &&
   672→                score >= bestScore + JOINT_SHORT_OPENING_MARGIN &&
   673→                (!best || score > best.score)) {
   674→                best = {
   675→                    ...row,
   676→                    score: QuranDB._round4(score),
   677→                    raw_score: QuranDB._round4(raw),
   678→                    bonus: 0,
   679→                    _prefix_rescue: true,
   680→                };
   681→            }
   682→        }
   683→        return best;
   684→    }

   846→    private _continuationBonuses(hint: [number, number] | null): Map<string, number> {
   847→        const bonuses = new Map<string, number>();
   848→        if (!hint)
   849→            return bonuses;
   850→        const [hSurah, hAyah] = hint;
   851→        const nv = this._byRef.get(`${hSurah}:${hAyah + 1}`);
   852→        if (nv) {
   853→            bonuses.set(`${hSurah}:${hAyah + 1}`, 0.22);
   854→            if (this._byRef.has(`${hSurah}:${hAyah + 2}`))
   855→                bonuses.set(`${hSurah}:${hAyah + 2}`, 0.12);
   856→            if (this._byRef.has(`${hSurah}:${hAyah + 3}`))
   857→                bonuses.set(`${hSurah}:${hAyah + 3}`, 0.06);
   858→        }
   859→        else {
   860→            const nextVerses = this._bySurah.get(hSurah + 1) ?? [];
   861→            const bonusValues = [0.22, 0.12, 0.06];
   862→            for (let i = 0; i < Math.min(nextVerses.length, 3); i++) {
   863→                bonuses.set(`${nextVerses[i].surah}:${nextVerses[i].ayah}`, bonusValues[i]);
   864→            }
   865→        }
   866→        return bonuses;

   868→    private static _suffixPrefixScore(text: string, verseText: string): number {
   869→        const wordsT = text.split(" ");
   870→        const wordsV = verseText.split(" ");
   871→        if (wordsT.length < 2 || wordsV.length < 2)
   872→            return 0.0;
   873→        let best = 0.0;
   874→        const maxTrim = Math.min(Math.floor(wordsT.length / 2), 4);
   875→        for (let trim = 1; trim <= maxTrim; trim++) {
   876→            const suffix = wordsT.slice(trim).join(" ");
   877→            const n = wordsT.length - trim;
   878→            const prefix = wordsV.slice(0, Math.min(n, wordsV.length)).join(" ");
   879→            best = Math.max(best, ratio(suffix, prefix));
   880→        }
   881→        return best;
   882→    }
   883→    private static _jointNgrams(s: string, n: number): Set<string> {

   892→    private static _intersectionSize(a: Set<string>, b: Set<string>): number {

   896→                count++;
