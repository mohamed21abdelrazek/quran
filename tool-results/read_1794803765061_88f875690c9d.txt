    79→    79→  /// Computed age from birthday. Falls back to stored age for legacy profiles.
    80→    80→  int get age {
    81→    81→    if (birthday == null) return _ageFallback;
    82→    82→    final now = DateTime.now();
    83→    83→    int years = now.year - birthday!.year;
    84→    84→    if (now.month < birthday!.month ||
    85→    85→        (now.month == birthday!.month && now.day < birthday!.day)) {
    86→    86→      years--;
    87→    87→    }
    88→    88→    return years.clamp(7, 100);
    89→    89→  }
    90→    90→
    91→    91→  /// Static helper to calculate age from a birthday for use before profile creation.
    92→    92→  static int calculateAge(DateTime birthday) {
    93→    93→    final now = DateTime.now();
    94→    94→    int years = now.year - birthday.year;
    95→    95→    if (now.month < birthday.month ||
    96→    96→        (now.month == birthday.month && now.day < birthday.day)) {
    97→    97→      years--;
    98→    98→    }
    99→    99→    return years.clamp(0, 120);
   100→   100→  }
